    42→class FakePaperEngine extends PaperEngine {
    43→  marks = new Map<string, number>();
    44→  constructor() {
    45→    super(TREND_TRADE.ACCOUNT_NAME);
    46→  }
    47→  private priceOf(symbol: string): number {
    48→    return this.marks.get(symbol) ?? 0;
    49→  }
    50→
    51→  override async open(input: {
    52→    symbol: string;
    53→    side: 'LONG' | 'SHORT';
    54→    leverage: number;
    55→    margin: number;
    56→  }) {
    57→    const price = this.priceOf(input.symbol);
    58→    if (price <= 0) throw new PaperError('未知交易对或行情未就绪');
    59→    const acc = await db.paperAccount.findUnique({ where: { name: TREND_TRADE.ACCOUNT_NAME } });
    60→    if (!acc) throw new PaperError('账户未初始化');
    61→    const fee = input.margin * input.leverage * TAKER_FEE;
    62→    if (input.margin + fee > acc.balance) {
    63→      throw new PaperError(`可用余额不足：需 ${input.margin.toFixed(2)} + 手续费 ${fee.toFixed(2)}`);
    64→    }
    65→    const notional = input.margin * input.leverage;
    66→    const qty = notional / price;
    67→    const liqPrice = liqPriceFor({
    68→      symbol: input.symbol,
    69→      side: input.side,
    70→      entry: price,
    71→      leverage: input.leverage,
    72→      notional,
    73→    });
    74→    const [pos] = await db.$transaction([
    75→      db.paperPosition.create({
    76→        data: {
    77→          accountId: acc.id,
    78→          symbol: input.symbol,
    79→          base: input.symbol.replace(/USDT$/, ''),
    80→          side: input.side,
    81→          leverage: input.leverage,
    82→          margin: input.margin,
    83→          qty,
    84→          entryPrice: price,
    85→          liqPrice,
    86→          openFee: fee,
    87→        },
    88→      }),
    89→      db.paperAccount.update({
    90→        where: { id: acc.id },
    91→        data: { balance: { decrement: input.margin + fee } },
    92→      }),
    93→      db.paperTrade.create({
    94→        data: {
    95→          accountId: acc.id,
    96→          symbol: input.symbol,
    97→          action: input.side === 'LONG' ? 'OPEN_LONG' : 'OPEN_SHORT',
    98→          price,
    99→          qty,
   100→          fee,
   101→          pnl: 0,
   102→        },
   103→      }),
   104→    ]);
   105→    return {
   106→      id: pos.id,
   107→      symbol: pos.symbol,
   108→      base: pos.base,
   109→      side: input.side,
   110→      leverage: pos.leverage,
   111→      margin: pos.margin,
   112→      qty: pos.qty,
   113→      entryPrice: pos.entryPrice,
   114→      markPrice: price,
   115→      liqPrice: pos.liqPrice,
   116→      notional,
   117→      unrealizedPnl: 0,
   118→      roePct: 0,
   119→      openedAt: pos.openedAt.toISOString(),
   120→    };
   121→  }
   122→
   123→  override async close(positionId: string) {
   124→    const pos = await db.paperPosition.findUnique({ where: { id: positionId } });
   125→    if (!pos || pos.status !== 'OPEN') throw new PaperError('该仓位已平仓或已清算');
   126→    const mark = this.priceOf(pos.symbol);
   127→    if (mark <= 0) throw new PaperError('行情暂不可用，请稍后再试');
   128→    const sign = pos.side === 'LONG' ? 1 : -1;
   129→    const gross = (mark - pos.entryPrice) * pos.qty * sign;
   130→    const closeFee = mark * pos.qty * TAKER_FEE;
   131→    const netPnl = gross - closeFee - pos.openFee;
   132→    await db.$transaction([
   133→      db.paperPosition.update({
   134→        where: { id: pos.id },
   135→        data: { status: 'CLOSED', exitPrice: mark, pnl: netPnl, closeFee, closedAt: new Date() },
   136→      }),
   137→      db.paperAccount.update({
   138→        where: { id: pos.accountId },
   139→        data: { balance: { increment: pos.margin + gross - closeFee } },
   140→      }),
   141→      db.paperTrade.create({
   142→        data: {
   143→          accountId: pos.accountId,
   144→          positionId: pos.id,
   145→          symbol: pos.symbol,
   146→          action: pos.side === 'LONG' ? 'CLOSE_LONG' : 'CLOSE_SHORT',
   147→          price: mark,
   148→          qty: pos.qty,
   149→          fee: closeFee,
   150→          pnl: netPnl,
   151→        },
   152→      }),
   153→    ]);
   154→    return { pnl: netPnl, symbol: pos.symbol, side: pos.side, price: mark };
   155→  }
   156→
   157→  override async liquidationCheck(accountId: string): Promise<void> {
   158→    const open = await db.paperPosition.findMany({ where: { accountId, status: 'OPEN' } });
   159→    for (const p of open) {
   160→      const mark = this.priceOf(p.symbol);
   161→      if (mark <= 0) continue;
   162→      const hit = p.side === 'LONG' ? mark <= p.liqPrice : mark >= p.liqPrice;
   163→      if (!hit) continue;
   164→      await db.paperPosition.update({
   165→        where: { id: p.id },
   166→        data: { status: 'LIQUIDATED', exitPrice: p.liqPrice, pnl: -p.margin, closedAt: new Date() },
   167→      });
   168→      await db.paperTrade.create({
   169→        data: {
   170→          accountId,
   171→          positionId: p.id,
   172→          symbol: p.symbol,
   173→          action: 'LIQUIDATE',
   174→          price: p.liqPrice,
   175→          qty: p.qty,
   176→          fee: 0,
   177→          pnl: -p.margin,
   178→        },
   179→      });
   180→    }
   181→  }
   182→}

   186→let barClock = Date.now() - 300_000; // 近真实时钟（先平后开独立性用真实平仓时刻比较）
   187→
   188→function sig(partial: {
   189→  symbol: string;
   190→  phase: string;
   191→  direction?: 'up' | 'down' | 'flat';
   192→  cycleId?: string | null;
   193→  dataFresh?: boolean;
   194→  score?: number;
   195→  lastBarTime?: number | null;
   196→}): UnifiedTrendSignal {
   197→  barClock += 60_000;
   198→  const dir =
   199→    partial.direction ??
   200→    (partial.phase === 'CONFIRMED_UP' ? 'up' : partial.phase === 'CONFIRMED_DOWN' ? 'down' : 'flat');
   201→  const cycleDir = partial.phase === 'CONFIRMED_UP' ? 'L' : partial.phase === 'CONFIRMED_DOWN' ? 'S' : 'X';
   202→  return {
   203→    symbol: partial.symbol,
   204→    market: 'futures',
   205→    interval: '1m',
   206→    phase: partial.phase as UnifiedTrendSignal['phase'],
   207→    direction: dir,
   208→    trendCycleId:
   209→      partial.cycleId !== undefined ? partial.cycleId : `${partial.symbol}:${cycleDir}:${barClock}`,
   210→    trendStartedAt: partial.phase.startsWith('CONFIRMED') ? barClock : null,
   211→    endPendingSince: null,
   212→    endEvidenceBars: 0,
   213→    endedAt: partial.phase === 'ENDED' ? barClock : null,
   214→    endReason: partial.phase === 'ENDED' ? 'DECAY_CONFIRMED' : null,
   215→    score: partial.score ?? 60,
   216→    strength: 0.6,
   217→    confirmCount: 3,
   218→    features: {
   219→      atr: 1, atrPct: 1.2, realizedVolPct: 40, volPercentile: 0.7, volRegime: 'HIGH',
   220→      efficiencyRatio: 0.5, upperWickPct: 0.1, lowerWickPct: 0.1, breakout: 'none',
   221→      falseBreakout: false, barsUsed: 480,
   222→    },
   223→    htf5m: { dir, efficiencyRatio: 0.5, returnPct: 1, valid: true, bars: 96 },
   224→    htf15m: { dir, efficiencyRatio: 0.5, returnPct: 1, valid: true, bars: 32 },
   225→    htfAgree: true,
   226→    lastBarTime: partial.lastBarTime !== undefined ? partial.lastBarTime : barClock,
   227→    lastKlineAt: barClock,
   228→    dataFresh: partial.dataFresh ?? true,
   229→    dataUnavailableReason: partial.dataFresh ?? true ? null : '测试：行情过期',
   230→    anomaly: { dir: null, level: null, priority: null, active: false, lastAt: null },
   231→    conflict: false,
   232→    reason: '测试信号',
   233→    changedAt: barClock,
   234→  } as UnifiedTrendSignal;
   235→}

   237→/** 保留原信号其余字段（测试内替换相位用） */
   238→function keepSig(s: UnifiedTrendSignal): UnifiedTrendSignal {
   239→  return { ...s };
   240→}
