   102→/** 找到时间 ≥ target 的最早采样点，计算相对当前价的涨跌 % */
   103→function chgSince(hist: PricePoint[], target: number, price: number): number | null {
   104→  if (!hist || hist.length === 0 || price <= 0) return null;
   105→  for (const pt of hist) {
   106→    if (pt.t >= target && pt.p > 0) return ((price - pt.p) / pt.p) * 100;
   107→  }
   108→  const lastPt = hist[hist.length - 1];
   109→  return lastPt.p > 0 ? ((price - lastPt.p) / lastPt.p) * 100 : null;
   110→}

   187→  private async boot(): Promise<void> {
   188→    // 1) 首拉合约清单（继承自旧实例的清单立即可用；缺失的源正常拉取）
   189→    await Promise.allSettled(FUTURES_SOURCE_PRIORITY.map((s) => this.refreshInstruments(s)));
   190→    // 2) 首拉价格 + 24h（继承数据已就绪时本轮只做增量刷新）
   191→    await Promise.allSettled([this.pollAllPrices(), this.pollAllStats()]);
   192→    // 3) 定时器：5s 价格 / 10s 统计到期检查（币安 24hr 实际 60s，Bybit/Gate 30s）
   193→    this.priceTimer = setInterval(() => void this.pollAllPrices(), PRICE_INTERVAL_MS);
   194→    this.tfTimer = setInterval(() => void this.pollAllStats(), 10_000);
   195→    // 4) 启动回填 Top（chg1m/chg5m 即刻可用；已有充足价格史的标的自动跳过）
   196→    void this.backfillTop();

   199→  // ============ 合约清单（每小时自刷新，失败 60s 重试） ============
   200→
   201→  private async refreshInstruments(source: FuturesSourceName): Promise<void> {
   202→    const st = this.sources[source];
   203→    // Task 60：挂起期间不出站（418 封禁期内打 instruments 同样刷新累犯计数、延长封禁）
   204→    const now = Date.now();
   205→    if (now < st.suspendUntil) {
   206→      setTimeout(() => void this.refreshInstruments(source), st.suspendUntil - now + 5_000);
   207→      return;
   208→    }
   209→    try {
   210→      st.instruments = await fetchInstruments(source);
   211→      st.instrumentsAt = Date.now();
   212→      st.failStreak = 0;
   213→      if (Date.now() >= st.bannedUntil) {
   214→        st.suspendUntil = 0;
   215→        st.bannedUntil = 0;
   216→        st.lastError = '';
   217→      }
   218→      setTimeout(() => void this.refreshInstruments(source), INSTRUMENT_REFRESH_MS);
   219→    } catch (e) {
   220→      // 418/429 同样按 retry-after 长挂起（instruments 撞墙也是撞墙）
   221→      this.applyRateLimitSuspend(source, e);
   222→      st.lastError = errMsg(e);
   223→      const wait = Math.max(60_000, st.suspendUntil - Date.now() + 5_000);
   224→      setTimeout(() => void this.refreshInstruments(source), wait);
   225→    }
   226→  }

   228→  // ============ 价格轮询（每 5s 全源并行） ============
   229→
   230→  private async pollAllPrices(): Promise<void> {
   231→    const now = Date.now();
   232→    await Promise.allSettled(
   233→      FUTURES_SOURCE_PRIORITY.map(async (s) => {
   234→        const st = this.sources[s];
   235→        if (now < st.suspendUntil || st.instruments.size === 0) return;
   236→        try {
   237→          const rows = await fetchPrices(s, st.instruments);
   238→          this.ingestPrices(s, rows);
   239→          st.failStreak = 0;
   240→          // Task 60：封禁期内偶发成功（边缘节点执行不一致）不提前解锁，挂满到 bannedUntil
   241→          if (Date.now() >= st.bannedUntil) {
   242→            st.suspendUntil = 0;
   243→            st.bannedUntil = 0;
   244→            st.lastError = '';
   245→          }
   246→        } catch (e) {
   247→          this.noteFailure(s, e);
   248→        }
   249→      }),
   250→    );
   251→    this.rebuildMerged();
   252→  }

   254→  private ingestPrices(source: FuturesSourceName, rows: { symbol: string; price: number }[]) {
   255→    const st = this.sources[source];
   256→    const now = Date.now();
   257→    for (const r of rows) {
   258→      let hist = st.priceHist.get(r.symbol);
   259→      if (!hist) {
   260→        hist = [];
   261→        st.priceHist.set(r.symbol, hist);
   262→      }
   263→      hist.push({ t: now, p: r.price });
   264→      this.trimHist(hist, now);
   265→    }
   266→    st.lastPriceAt = now;
   267→  }
   268→
   269→  /** 压缩历史：20 分钟内保留 5s 精度，更早的压缩为 1 分钟精度，总量 ≤ 75 分钟 */
   270→  private trimHist(hist: PricePoint[], now: number) {
   271→    const maxAge = now - 75 * 60_000;
   272→    let drop = 0;
   273→    while (drop < hist.length && hist[drop].t < maxAge) drop++;
   274→    if (drop > 0) hist.splice(0, drop);
   275→    const cutoff = now - 20 * 60_000;
   276→    let bound = 0;
   277→    while (bound < hist.length && hist[bound].t < cutoff) bound++;
   278→    if (bound <= 1) return;
   279→    const compacted: PricePoint[] = [];
   280→    let lastMinute = -1;
   281→    for (let i = 0; i < bound; i++) {
   282→      const m = Math.floor(hist[i].t / 60_000);
   283→      if (m !== lastMinute) {
   284→        compacted.push(hist[i]);
   285→        lastMinute = m;
   286→      }
   287→    }
   288→    if (compacted.length < bound) {
   289→      hist.splice(0, bound, ...compacted);
   290→    }
   291→  }

   293→  // ============ 24hr 统计（每 30s 全源并行，差分 vol1m） ============
   294→
   295→  private async pollAllStats(): Promise<void> {
   296→    const now = Date.now();
   297→    await Promise.allSettled(
   298→      FUTURES_SOURCE_PRIORITY.map(async (s) => {
   299→        const st = this.sources[s];
   300→        if (now < st.suspendUntil || st.instruments.size === 0) return;
   301→        if (now < st.nextStatsAt) return; // 每源差异化间隔（币安 60s，其余 30s）
   302→        st.nextStatsAt = now + STATS_INTERVAL_MS[s];
   303→        try {
   304→          const rows = await fetch24h(s, st.instruments);
   305→          this.ingestStats(s, rows, Date.now());
   306→          st.failStreak = 0;
   307→          // Task 60：封禁期内偶发成功不提前解锁（同价格轮询口径）
   308→          if (Date.now() >= st.bannedUntil) {
   309→            st.suspendUntil = 0;
   310→            st.bannedUntil = 0;
   311→            st.lastError = '';
   312→          }
   313→        } catch (e) {
   314→          st.nextStatsAt = now + STATS_RETRY_MS; // 失败快速重试
   315→          this.noteFailure(s, e);
   316→        }
   317→      }),
   318→    );
   319→    this.rebuildMerged();
   320→  }

   322→  private ingestStats(source: FuturesSourceName, rows: Stats24Row[], now: number) {
   323→    const st = this.sources[source];
   324→    for (const row of rows) {
   325→      // 差分出每分钟成交额（窗口下限仅防零间隔抖动，同现货 collector 参数）
   326→      const prev = st.lastVolSnap.get(row.symbol);
   327→      if (prev) {
   328→        const dtMin = (now - prev.at) / 60_000;
   329→        if (dtMin >= 0.35 && dtMin <= 6) {
   330→          const perMin = (row.qv24 - prev.qv24) / dtMin;
   331→          if (Number.isFinite(perMin) && perMin >= 0) {
   332→            const bucket = Math.floor(now / 60_000) * 60_000;
   333→            let vs = st.volSamples.get(row.symbol);
   334→            if (!vs) {
   335→              vs = [];
   336→              st.volSamples.set(row.symbol, vs);
   337→            }
   338→            const lastS = vs[vs.length - 1];
   339→            if (lastS && lastS.t === bucket) lastS.qv = perMin;
   340→            else {
   341→              vs.push({ t: bucket, qv: perMin });
   342→              if (vs.length > 30) vs.splice(0, vs.length - 30);
   343→            }
   344→          }
   345→        }
   346→      }
   347→      st.lastVolSnap.set(row.symbol, { at: now, qv24: row.qv24 });
   348→      st.stats24.set(row.symbol, row);
   349→    }
   350→    st.last24hAt = now;
   351→  }

   353→  /** Task 60：418/429 封禁挂起（挂满服务方告知的整个封禁期）。
   354→   *  命中返回 true（调用方跳过常规失败计数）；未命中返回 false。
   355→   *  ⚠️ 本方法与「成功路径清挂起」的协作语义：封禁期内偶发成功（币安边缘节点
   356→   *  封禁执行不一致）不得解锁——见各成功路径的 bannedUntil 检查。 */
   357→  private applyRateLimitSuspend(source: FuturesSourceName, e: unknown): boolean {
   358→    const msg = errMsg(e);
   359→    if (!msg.includes('HTTP 418') && !msg.includes('HTTP 429')) return false;
   360→    const st = this.sources[source];
   361→    const ra = e instanceof FuturesFetchError ? e.retryAfterMs : 0;
   362→    const until =
   363→      Date.now() +
   364→      Math.min(Math.max(ra, RATE_LIMIT_SUSPEND_MS) + 60_000, RATE_LIMIT_BAN_CAP_MS);
   365→    st.suspendUntil = Math.max(st.suspendUntil, until);
   366→    st.bannedUntil = Math.max(st.bannedUntil, until);
   367→    st.failStreak = 0;
   368→    st.lastError = msg;
   369→    return true;
   370→  }
   371→
   372→  private noteFailure(source: FuturesSourceName, e: unknown) {
   373→    const st = this.sources[source];
   374→    const msg = errMsg(e);
   375→    st.lastError = msg;
   376→    // 418/429 = 限频/IP 封禁：按服务方 retry-after 挂满封禁期（封禁期内零出站，
   377→    // 避免反复撞墙刷新币安累犯计数、把封禁时长越顶越高）
   378→    if (this.applyRateLimitSuspend(source, e)) return;
   379→    st.failStreak++;
   380→    if (st.failStreak >= FAIL_STREAK_LIMIT) {
   381→      st.suspendUntil = Date.now() + SUSPEND_MS;
   382→      st.failStreak = 0;
   383→    }
   384→  }
   385→
   386→  // ============ 分片：按成交额把品种分配给三源（boot 后首次统计就绪时分配一次） ============
   387→
   388→  /** 排序权重：优先币安自身 24h 成交额（供价基准一致），不可用时退其它源（口径差异不影响排序大局） */
   389→  private rankKey(symbol: string): number {
   390→    for (const s of FUTURES_SOURCE_PRIORITY) {
   391→      const st = this.sources[s].stats24.get(symbol);
   392→      if (st && st.qv24 > 0) return st.qv24;
   393→    }
   394→    return 0;
   395→  }
   396→
   397→  private assignShards(): void {
   398→    if (this.shards.size > 0) return; // 运行期不重排：避免 K 线序列跨源拼接
   399→    // 等统计就绪（任意源有 24h 成交额即可排序；币安被封时 Bybit/Gate 统计照常流入）
   400→    const hasStats = FUTURES_SOURCE_PRIORITY.some(
   401→      (s) => this.sources[s].stats24.size > 0,
   402→    );
   403→    if (!hasStats || this.sources.binance.instruments.size === 0) return;
   404→
   405→    const whitelist = this.sources.binance.instruments;
   406→    const sorted = [...whitelist.keys()]
   407→      .filter((sym) => !isExcludedBase(whitelist.get(sym)!.base))
   408→      .sort((a, b) => this.rankKey(b) - this.rankKey(a));
   409→
   410→    let otherToggle = 0; // bybit/gate 交替发牌 → 两所成交额质量大体对半
   411→    const quota: Record<FuturesSourceName, number> = {
   412→      binance: BINANCE_SHARD_SIZE,
   413→      bybit: OTHER_SHARD_QUOTA,
   414→      gate: OTHER_SHARD_QUOTA,
   415→    };
   416→    for (const sym of sorted) {
   417→      // 币安配额内（Top125）：按优先级+配额选归属
   418→      let home: FuturesSourceName | null = null;
   419→      for (const s of FUTURES_SOURCE_PRIORITY) {
   420→        if (quota[s] <= 0 || !this.sources[s].instruments.has(sym)) continue;
   421→        home = s;
   422→        quota[s]--;
   423→        break;
   424→      }
   425→      // 配额用尽 → 双所覆盖则交替发牌；单所覆盖归该所；两所都不覆盖（新上币等）回币安
   426→      if (!home) {
   427→        const covered = FUTURES_SOURCE_PRIORITY.filter(
   428→          (s) => s !== 'binance' && this.sources[s].instruments.has(sym),
   429→        );
   430→        if (covered.length === 2) {
   431→          home = covered[otherToggle % 2];
   432→          otherToggle++;
   433→        } else if (covered.length === 1) {
   434→          home = covered[0];
   435→        } else {
   436→          home = 'binance'; // 极少数（如刚上线币安的新品种）：币安兜底，K 线成本可忽略
   437→        }
   438→      }
   439→      this.shards.set(sym, home);
   440→    }
   441→    this.shardsAt = Date.now();
   442→  }

   444→  /** 归属源（未分片时按优先级首个覆盖源兑底） */
   445→  private homeOf(symbol: string): FuturesSourceName {
   446→    const home = this.shards.get(symbol);
   447→    if (home) return home;
   448→    for (const s of FUTURES_SOURCE_PRIORITY) {
   449→      if (this.sources[s].instruments.has(symbol)) return s;
   450→    }
   451→    return 'binance';
   452→  }

   454→  /** 为符号选择供价源：归属源优先，其余按优先级补位（须「未挂起 + 价格史新鲜」） */
   455→  private selectSource(symbol: string, now: number): FuturesSourceName | null {
   456→    const home = this.homeOf(symbol);
   457→    const order = [
   458→      home,
   459→      ...FUTURES_SOURCE_PRIORITY.filter((s) => s !== home),
   460→    ];
   461→    for (const s of order) {
   462→      const st = this.sources[s];
   463→      if (now < st.suspendUntil) continue;
   464→      if (now - st.lastPriceAt > PRICE_FRESH_MS) continue;
   465→      const hist = st.priceHist.get(symbol);
   466→      if (!hist || hist.length === 0) continue;
   467→      const last = hist[hist.length - 1];
   468→      if (now - last.t > PRICE_FRESH_MS || !(last.p > 0)) continue;
   469→      return s;
   470→    }
   471→    return null;
   472→  }

   474→  /** 重建聚合行情：白名单 = 币安 U 本位永续（只做币安包含的品种） */
   475→  private rebuildMerged(): void {
   476→    this.assignShards(); // 首次统计就绪时分配（幂等）
   477→    const now = Date.now();
   478→    const whitelist = this.sources.binance.instruments;
   479→    for (const [symbol, ins] of whitelist) {
   480→      if (isExcludedBase(ins.base)) {
   481→        this.merged.delete(symbol); // 稳定币对 / 代币化股票等（USDC 永续等）
   482→        continue;
   483→      }
   484→      const sel = this.selectSource(symbol, now);
   485→      if (!sel) {
   486→        this.merged.delete(symbol); // 全源无新鲜数据 → 移出（恢复后自动回来）
   487→        continue;
   488→      }
   489→      const st = this.sources[sel];
   490→      const hist = st.priceHist.get(symbol)!;
   491→      const lastPt = hist[hist.length - 1];
   492→      const price = lastPt.p;
   493→      const prevPt = hist.length >= 2 ? hist[hist.length - 2] : null;
   494→      const s24 = st.stats24.get(symbol);
   495→      // 24h 成交额跨源取最大：统一门槛/分层口径（Gate/Bybit 口径成交额系统性低于币安，
   496→      // 按供价源口径会误杀尾部标的）。vol1m/量比保持供价源内部口径（与 K 线基线同源自洽）。
   497→      let qv24 = s24?.qv24 ?? 0;
   498→      for (const src of FUTURES_SOURCE_PRIORITY) {
   499→        const cand = this.sources[src].stats24.get(symbol);
   500→        if (cand && cand.qv24 > qv24) qv24 = cand.qv24;
   501→      }
   502→      const vs = st.volSamples.get(symbol) ?? [];
   503→      let vol1m = 0;
   504→      let volAvg20m = 0;
   505→      if (vs.length >= 5) {
   506→        const window = vs.slice(-21, -1); // 过去 20 个完整分钟
   507→        volAvg20m = window.reduce((acc, x) => acc + x.qv, 0) / window.length;
   508→        vol1m = vs[vs.length - 1].qv;
   509→      }
   510→      const prevRow = this.merged.get(symbol);
   511→      const tickDir: 1 | -1 | 0 =
   512→        prevPt && price !== prevPt.p ? (price > prevPt.p ? 1 : -1) : (prevRow?.tickDir ?? 0);
   513→      const row: MergedRow = {
   514→        symbol,
   515→        base: ins.base,
   516→        price,
   517→        open: s24?.open24 || price,
   518→        high24h: s24?.high24 || price,
   519→        low24h: s24?.low24 || price,
   520→        change24h: s24?.chg24 ?? 0,
   521→        quoteVolume24h: qv24,
   522→        chg1m: chgSince(hist, now - 60_000, price),
   523→        chg5m: chgSince(hist, now - 5 * 60_000, price),
   524→        chg15m: chgSince(hist, now - 15 * 60_000, price),
   525→        chg1h: chgSince(hist, now - 60 * 60_000, price),
   526→        vol1m: vol1m > 0 ? vol1m : null,
   527→        volMultiple: volAvg20m > 0 && vol1m > 0 ? vol1m / volAvg20m : null,
   528→        tickDir,
   529→        hotTags: null,
   530→        updatedAt: st.lastPriceAt || now,
   531→        source: sel,
   532→      };
   533→      this.merged.set(symbol, row);
   534→    }
   535→  }

   600→  // ============ K 线（异动引擎基线 + 图表共用；按符号同源取数） ============
   601→
   602→  /** K 线取数顺序：归属源（分片）优先；兜底补位时非币安源优先，币安永远最后——
   603→   *  （Task 51：K 线不全走币安，币安只背它分片那份 + 两所同时不可用时的末位兑底）。
   604→   *  挂起中的源直接跳过——币安 418 封禁期间 K 线自动改走 Bybit/Gate，不在墙上硬撞 */
   605→  private klineSourceOrder(symbol: string): FuturesSourceName[] {
   606→    const now = Date.now();
   607→    const order: FuturesSourceName[] = [];
   608→    const home = this.homeOf(symbol);
   609→    if (now >= this.sources[home].suspendUntil) order.push(home);
   610→    const others = FUTURES_SOURCE_PRIORITY.filter((s) => s !== home).sort(
   611→      (a, b) => (a === 'binance' ? 1 : 0) - (b === 'binance' ? 1 : 0),
   612→    );
   613→    for (const s of others) {
   614→      if (now < this.sources[s].suspendUntil) continue;
   615→      if (!this.sources[s].instruments.has(symbol)) continue;
   616→      order.push(s);
   617→    }
   618→    return order;
   619→  }

   621→  /** 取 K 线并告知实际来源（回填播种需要同源写入价格史，保持「单源供价」不变式） */
   622→  async fetchKlinesFrom(
   623→    symbol: string,
   624→    interval: '1m' | '5m' | '15m' | '30m' | '1h' | '4h' | '1d',
   625→    opts: KlinesOpts,
   626→  ): Promise<{ source: FuturesSourceName; candles: FuturesCandle[] }> {
   627→    for (const s of this.klineSourceOrder(symbol)) {
   628→      const ins = this.sources[s].instruments.get(symbol);
   629→      if (!ins) continue;
   630→      this.sources[s].klineReqs++; // 分片分配观测：实际承接的 K 线请求计数
   631→      try {
   632→        const candles = await fetchKlines(s, ins.native, interval, opts);
   633→        if (candles.length > 0) return { source: s, candles };
   634→      } catch {
   635→        // 换下一源
   636→      }
   637→    }
   638→    throw new Error(`合约 K 线不可用：${symbol}`);
   639→  }

   641→  async fetchKlinesFor(
   642→    symbol: string,
   643→    interval: '1m' | '5m' | '15m' | '30m' | '1h' | '4h' | '1d',
   644→    opts: KlinesOpts,
   645→  ): Promise<FuturesCandle[]> {
   646→    return (await this.fetchKlinesFrom(symbol, interval, opts)).candles;
   647→  }

   649→  // ============ 启动回填：Top120 的 1m K 线 → chg1m/chg5m 即刻可用 ============
   650→
   651→  private async backfillTop(): Promise<void> {
   652→    try {
   653→      for (let i = 0; i < 30 && this.merged.size === 0; i++) await sleep(1000);
   654→      if (this.merged.size === 0) return;
   655→      // 已有 ≥20 个价格点的标的跳过（bump key 交接/HMR 后不重复回填）
   656→      const top = [...this.merged.values()]
   657→        .filter((r) => {
   658→          if (r.quoteVolume24h < 1_000_000) return false;
   659→          const hist = this.sources[r.source]?.priceHist.get(r.symbol);
   660→          return !hist || hist.length < 20;
   661→        })
   662→        .sort((a, b) => b.quoteVolume24h - a.quoteVolume24h)
   663→        .slice(0, TOP_BACKFILL);
   664→      if (top.length === 0) return;
   665→      const now = Date.now();
   666→      let cursor = 0;
   667→      const worker = async () => {
   668→        while (cursor < top.length) {
   669→          const row = top[cursor++];
   670→          try {
   671→            const res = await this.fetchKlinesFrom(row.symbol, '1m', { limit: 75 });
   672→            this.seedHistory(row, res.source, res.candles, now);
   673→          } catch {
   674→            // 单币种失败忽略
   675→          }
   676→          await sleep(300); // 温和节流：与引擎回填错峰，防币安 fapi 418 限频
   677→        }
   678→      };
   679→      await Promise.all(Array.from({ length: 2 }, worker)); // 并发 2（Task 50 节流）
   680→    } catch {
   681→      // 回填失败不影响后续实时采集
   682→    }
   683→  }

   685→  /** 播种价格史/量能基线——只写「实际供 K 线的那个源」，避免跨源拼接 */
   686→  private seedHistory(
   687→    row: MergedRow,
   688→    source: FuturesSourceName,
   689→    candles: FuturesCandle[],
   690→    now: number,
   691→  ) {
   692→    if (source !== row.source) return; // 供价源 K 线失败且降级 → 不播种（等实时累积）
   693→    const st = this.sources[source];
   694→    const hist: PricePoint[] = candles
   695→      .filter((c) => Number.isFinite(c.c) && c.c > 0)
   696→      .map((c) => ({ t: c.t, p: c.c }));
   697→    if (hist.length > 0) {
   698→      hist.push({ t: now, p: row.price });
   699→      st.priceHist.set(row.symbol, hist);
   700→    }
   701→    const vs: { t: number; qv: number }[] = [];
   702→    for (const c of candles) {
   703→      if (c.t >= now - 21 * 60_000 && c.t + 60_000 <= now && Number.isFinite(c.qv)) {
   704→        vs.push({ t: c.t, qv: c.qv });
   705→      }
   706→    }
   707→    if (vs.length >= 4) st.volSamples.set(row.symbol, vs);
   708→  }
   709→}
