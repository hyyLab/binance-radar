     1→// ---------- AutoPaperTrader：趋势自动模拟交易服务（Task 54 §3/§四 / Task 56 用户定版） ----------
     2→// 消费 TrendWatchService 输出的统一趋势信号（UnifiedTrendSignal，单一趋势来源——
     3→// 本服务绝不另行计算趋势方向），按固定规则开仓/持仓/平仓；纯模拟，不接入真实下单。
     4→//
     5→// 交易规则（Task 56 用户定版）：
     6→// - 独立账户 trend-auto-v1（PaperAccount.name 隔离；手动 default 盘完全不受影响）
     7→// - 开仓：确认趋势（CONFIRMED_UP/DOWN）+ 质量门槛（score/ER）+ 高周期过滤 + 二次确认
     8→//   （首次确认只建待入场记录，等下一根已收盘 K 线复核后开仓）+ 同币种冷却 5 分钟；
     9→//   每仓保证金 = 开仓时账户权益 × min(marginRatio, 95%÷持仓数)（Task 58：持仓数多时
    10→//   自动压缩单仓比例保证满仓可开，≤4 仓行为与原 20% 定版一致），杠杆固定 20x；
    11→//   保证金+手续费超出可用余额直接跳过并记录（不变相提高比例/杠杆）；
    12→//   同一趋势周期只允许一笔（决策键幂等），不允许加仓摊平
    13→// - 离场：不设固定止盈止损——趋势结束必须满足「连续 END_CONFIRM_BARS 根不同且有效的
    14→//   已收盘 K 线持续出现结束证据」并由 lifecycle 判定 ENDED 后才离场，且 ENDED 必须
    15→//   归属于持仓绑定的同一 trendCycleId（旧周期 ENDED 不得误杀新周期持仓，§2.2）、
    16→//   证据 K 线必须晚于开仓依据的 K 线（§2.3）；END_PENDING 仅记录首次报警时刻，不平仓；
    17→//   反向趋势独立确认（dataFresh）可立即平掉原方向持仓；保留强平机制（独立巡检任务）
    18→// - 数据不可用（DATA_UNAVAILABLE/过期）：不新开仓、不伪装趋势结束、继续按标记价
    19→//   管理强平；暂停模式：只停新开仓（含待入场转换），继续退出管理与强平巡检
    20→// - 成本口径：开/平仓手续费双边各 0.02%（Task 59 用户定版，TREND_TRADE.FEE_RATE；
    21→//   经 PaperEngine 实例属性 feeRate 生效——手动盘 default 账户仍为默认 Taker 0.05%）；
    22→//   滑点按可配基点估算，成交记录展示；统计主口径 = 含滑点净盈亏（更保守，不省略成本）
    23→// - 持仓结果采集（Task 56 §七）：MFE/MAE 按主循环可用标记价增量更新并持久化，
    24→//   绝不用事后最高/最低价构造决策时不存在的数据
    25→// - 成交假设：以决策时刻标记价成交（非信号 K 线收盘价），绝不使用未来行情
    26→//
    27→// 生命周期：绑定 EnhancedEngine（boot 拉起 / stop 停表）；单例由引擎单例保证；
    28→// 重启后从 DB 恢复持仓关联（TrendTradeMeta）、待入场记录（TrendPendingEntry）并 reconcile。

   154→  // ============ 生命周期 ============
   155→
   156→  async ensureStarted(): Promise<void> {
   157→    if (this.started) return;
   158→    if (this.starting) {
   159→      await Promise.race([this.starting, new Promise((r) => setTimeout(r, 6000))]);
   160→      return;
   161→    }
   162→    this.started = true;
   163→    this.stopped = false;
   164→    this.startedAt = Date.now();
   165→    this.starting = this.boot().finally(() => {
   166→      this.starting = null;
   167→    });
   168→    await Promise.race([this.starting, new Promise((r) => setTimeout(r, 6000))]);
   169→  }
   170→
   171→  private async boot(): Promise<void> {
   172→    // 1) 建表（幂等，含 Task 56 扩展列/待入场表）+ 播种配置 + 独立账户初始化
   173→    await ensureTrendTradeSchema();
   174→    const seeded = await this.repo.loadConfig(TREND_TRADE.ACCOUNT_NAME);
   175→    if (!seeded) {
   176→      await this.repo.seedConfig({
   177→        name: TREND_TRADE.ACCOUNT_NAME,
   178→        mode: 'RUNNING',
   179→        initialCapital: TREND_TRADE.INITIAL_CAPITAL,
   180→        marginRatio: TREND_TRADE.MARGIN_RATIO,
   181→        leverage: TREND_TRADE.LEVERAGE,
   182→        maxPositions: TREND_TRADE.MAX_POSITIONS,
   183→        slippageBps: TREND_TRADE.SLIPPAGE_BPS,
   184→        peakEquity: 0,
   185→      });
   186→    }
   187→    const acc = await db.paperAccount.findUnique({ where: { name: TREND_TRADE.ACCOUNT_NAME } });
   188→    if (!acc) {
   189→      await db.paperAccount.create({
   190→        data: { name: TREND_TRADE.ACCOUNT_NAME, balance: TREND_TRADE.INITIAL_CAPITAL },
   191→      });
   192→    }
   193→    // 2) 装配策略（Task 57：DB 配置的 strategyId + 恢复策略轮次状态；外部注入的固定策略不覆盖）
   194→    if (!this.fixedStrategy) {
   195→      const cfgNow = await this.repo.loadConfig(TREND_TRADE.ACCOUNT_NAME);
   196→      await this.applyStrategy(cfgNow?.strategyId ?? TREND_TRADE.DEFAULT_STRATEGY_ID);
   197→    } else {
   198→      this.strategy.bindPersistence((sid, sym, json) => {
   199→        void this.repo.saveStrategyState(sid, sym, json).catch(() => {});
   200→      });
   201→    }
   202→    // 3) reconcile：修复崩溃窗口（开仓成功但 meta 未写 / 平仓成功但 meta 未关）
   203→    await this.reconcile();
   204→    // 4) 主循环（独立持续风险检查任务：无页面访问也照常巡检强平与管理持仓）
   205→    if (!this.noTimer) this.timer = setInterval(() => void this.tick(), TREND_TRADE.SCAN_MS);
   206→  }
   207→
   208→  /**
   209→   * 装配策略实例（boot / 运行期切换 / tick 自愈 / 控制台调参后共用）：创建实例（带 DB
   210→   * 参数覆盖——Task 57-c 控制台）、绑定状态持久化、从 TrendStrategyState 恢复轮次状态。
   211→   * 参数解析失败（脏数据）回退代码默认，绝不阻断交易循环。
   212→   */
   213→  private async applyStrategy(strategyId: string): Promise<void> {
   214→    let params: Record<string, unknown> | undefined;
   215→    try {
   216→      const cfgNow = await this.repo.loadConfig(TREND_TRADE.ACCOUNT_NAME);
   217→      const raw = parseStrategyParamsMap(cfgNow?.strategyParamsJson)[strategyId];
   218→      if (raw && typeof raw === 'object') {
   219→        params = sanitizeRuntimeParamsLenient(raw).strategyParams as Record<string, unknown>;
   220→      }
   221→    } catch {
   222→      params = undefined; // 配置读取失败回退代码默认
   223→    }
   224→    const s = createStrategy(strategyId, params);
   225→    s.bindPersistence((sid, sym, json) => {
   226→      void this.repo.saveStrategyState(sid, sym, json).catch(() => {
   227→        // 状态持久化失败不影响决策（重启后轮次可能回退一档，可接受）
   228→      });
   229→    });
   230→    try {
   231→      s.importState(await this.repo.loadStrategyStates(s.id));
   232→    } catch {
   233→      // 恢复失败从空状态开始（轮次按首次处理，保守不丢单）
   234→    }
   235→    this.strategy = s;
   236→  }
   237→
   238→  /**
   239→   * 切换交易算法（Task 57 UI 入口）：持久化 strategyId + 装配新实例。
   240→   * 持仓清场由调用方（control API）先执行——不同算法持仓管理规则不同，不混管。
   241→   */
   242→  async setStrategyId(strategyId: string): Promise<void> {
   243→    await this.repo.updateStrategyId(TREND_TRADE.ACCOUNT_NAME, strategyId);
   244→    await this.applyStrategy(strategyId);
   245→  }
   246→
   247→  // ============ 参数控制台（Task 57-c：用户可自行调整自动开仓参数） ============
   248→
   249→  /**
   250→   * 保存运行时参数（控制台「保存」入口）并热生效：
   251→   * - strategyParams：当前策略的覆盖项（整体替换该 strategyId 的覆盖对象；传 {} = 全回默认）；
   252→   *   非固定注入策略时重装配实例（轮次状态从 DB 恢复，语义不丢）。
   253→   * - waitBarConfirm / cooldownMs / marginRatio：基础设施参数，auto-trader 每 tick 重读
   254→   *   配置，下一 tick 生效（marginRatio 实际比例 = min(设定值, 95%÷持仓数)，见 config.ts）。
   255→   * 入参已在 control API 层严格校验（白名单 + 范围）。
   256→   */
   257→  async setRuntimeParams(patch: {
   258→    strategyParams?: Record<string, unknown>;
   259→    waitBarConfirm?: number;
   260→    cooldownMs?: number;
   261→    marginRatio?: number;
   262→  }): Promise<void> {
   263→    const name = TREND_TRADE.ACCOUNT_NAME;
   264→    const cfg = await this.repo.loadConfig(name);
   265→    const strategyId = cfg?.strategyId ?? TREND_TRADE.DEFAULT_STRATEGY_ID;
   266→    const dbPatch: {
   267→      strategyParamsJson?: string | null;
   268→      waitBarConfirm?: number;
   269→      cooldownMs?: number | null;
   270→      marginRatio?: number;
   271→    } = {};
   272→    let strategyChanged = false;
   273→    if (patch.strategyParams !== undefined) {
   274→      const map = parseStrategyParamsMap(cfg?.strategyParamsJson);
   275→      map[strategyId] = patch.strategyParams;
   276→      dbPatch.strategyParamsJson = JSON.stringify(map);
   277→      strategyChanged = true;
   278→    }
   279→    if (patch.waitBarConfirm !== undefined) dbPatch.waitBarConfirm = patch.waitBarConfirm;
   280→    if (patch.cooldownMs !== undefined) dbPatch.cooldownMs = patch.cooldownMs;
   281→    if (patch.marginRatio !== undefined) dbPatch.marginRatio = patch.marginRatio;
   282→    if (Object.keys(dbPatch).length === 0) return;
   283→    await this.repo.updateRuntimeParams(name, dbPatch);
   284→    if (strategyChanged && !this.fixedStrategy) {
   285→      await this.applyStrategy(strategyId); // 热装配：新参数立即用于后续评估；轮次状态从 DB 恢复
   286→      logDecision(`参数控制台：策略参数已更新并热应用（strategy=${strategyId} params=${JSON.stringify(patch.strategyParams)}）`);
   287→    }
   288→    // marginRatio 为基础设施列：每 tick 重读 cfg，下一 tick 生效（无需重装配）
   289→    if (patch.marginRatio !== undefined) {
   290→      logDecision(`参数控制台：每仓保证金比例已更新为 ${patch.marginRatio}（下一 tick 生效；实际比例 = min(设定值, 95%÷持仓数)）`);
   291→    }
   292→  }
   293→
   294→  /**
   295→   * 恢复全部控制台参数为代码默认（控制台「恢复默认」入口）：
   296→   * 清除当前策略的参数覆盖 + 二次确认开关回开 + 冷却回默认 + 保证金比例回 20%；轮次状态保留。
   297→   */
   298→  async resetRuntimeParams(): Promise<void> {
   299→    const name = TREND_TRADE.ACCOUNT_NAME;
   300→    const cfg = await this.repo.loadConfig(name);
   301→    const strategyId = cfg?.strategyId ?? TREND_TRADE.DEFAULT_STRATEGY_ID;
   302→    const map = parseStrategyParamsMap(cfg?.strategyParamsJson);
   303→    delete map[strategyId];
   304→    await this.repo.updateRuntimeParams(name, {
   305→      strategyParamsJson: JSON.stringify(map),
   306→      waitBarConfirm: 1,
   307→      cooldownMs: null,
   308→      marginRatio: TREND_TRADE.MARGIN_RATIO, // 列 NOT NULL，回默认值而非 null
   309→    });
   310→    if (!this.fixedStrategy) {
   311→      await this.applyStrategy(strategyId);
   312→      logDecision(`参数控制台：已恢复全部参数为代码默认（strategy=${strategyId}）`);
   313→    }
   314→  }

   316→  stop(): void {
   317→    this.stopped = true;
   318→    this.started = false;
   319→    if (this.timer) clearInterval(this.timer);
   320→    this.timer = null;

   323→  /**
   324→   * 崩溃窗口修复（任务 §七：服务重启后必须恢复持仓关联信息）：
   325→   * a) 自动账户 OPEN 持仓缺 TrendTradeMeta → 从最近一条已抢占未绑定的 OPEN 决策恢复
   326→   *    cycleId；找不到则标 ORPHAN 周期（保守：只受强平/人工/反向确认管理——ORPHAN 周期
   327→   *    无从与 ENDED 事件做周期匹配，按 §2.2 不再「仅凭 symbol」离场）。
   328→   * b) meta OPEN 但 PaperPosition 已 CLOSED/LIQUIDATED → 补写平仓终态。
   329→   */
   330→  private async reconcile(): Promise<void> {
   331→    try {
   332→      const acc = await db.paperAccount.findUnique({ where: { name: TREND_TRADE.ACCOUNT_NAME } });
   333→      if (!acc) return;
   334→      const openPositions = await db.paperPosition.findMany({
   335→        where: { accountId: acc.id, status: 'OPEN' },
   336→      });
   337→      for (const p of openPositions) {
   338→        const meta = await this.repo.getMeta(p.id);
   339→        if (!meta) {
   340→          const claimed = await db.$queryRawUnsafe<{ cycleId: string; createdAt: Date }[]>(
   341→            `SELECT cycleId, createdAt FROM TrendAutoDecision
   342→             WHERE symbol = ? AND action = 'OPEN' AND positionId IS NULL
   343→             ORDER BY createdAt DESC LIMIT 1`,
   344→            p.symbol,
   345→          );
   346→          const cycleId = claimed[0]?.cycleId ?? `ORPHAN:${p.symbol}:${p.openedAt.getTime()}`;
   347→          await this.repo.upsertMeta({
   348→            positionId: p.id,
   349→            trendCycleId: cycleId,
   350→            symbol: p.symbol,
   351→            side: p.side,
   352→            trendPhaseAtOpen: 'CONFIRMED_RESTORED',
   353→            scoreAtOpen: 0,
   354→            featuresAtOpen: JSON.stringify({ restored: true, recoveredAt: Date.now() }),
   355→            signalAtOpen: '{}',
   356→            slippageEst: p.qty * p.entryPrice * (TREND_TRADE.SLIPPAGE_BPS / 10_000),
   357→            status: 'OPEN',
   358→            exitReason: null,
   359→            exitSignalJson: null,
   360→            netPnlWithSlip: null,
   361→            openDecisionAt: claimed[0]?.createdAt ?? p.openedAt,
   362→            closedAt: null,
   363→          });
   364→        }
   365→      }
   366→      // meta OPEN 但仓位已终态 → 补写
   367→      const openMetas = await this.repo.loadMetasByStatus('OPEN');
   368→      for (const m of openMetas) {
   369→        const pos = await db.paperPosition.findUnique({ where: { id: m.positionId } });
   370→        if (!pos) {
   371→          await this.repo.upsertMeta({ ...m, status: 'CLOSED', exitReason: 'MANUAL', closedAt: new Date() });
   372→          continue;
   373→        }
   374→        if (pos.status === 'LIQUIDATED') {
   375→          await this.finalizeLiquidated(m, pos.margin, pos.pnl ?? -pos.margin);
   376→        } else if (pos.status === 'CLOSED') {
   377→          await this.repo.upsertMeta({
   378→            ...m,
   379→            status: 'CLOSED',
   380→            exitReason: 'MANUAL',
   381→            netPnlWithSlip: (pos.pnl ?? 0) - m.slippageEst,
   382→            closedAt: pos.closedAt ?? new Date(),
   383→          });
   384→        }
   385→      }
   386→    } catch (e) {
   387→      this.lastError = e instanceof Error ? e.message : String(e);
   388→      console.warn('[trend-auto] reconcile 失败：', this.lastError);
   389→    }
   390→  }
   391→
   392→  // ============ 主循环 ============
   393→
   394→  /** 单步主循环：public 以便测试/回放驱动（生产由 SCAN_MS 定时器调用） */
   395→  // 重入保护：setInterval 不等待异步完成——上一轮 tick 未跑完（行情/DB 慢）时下一轮并发
   396→  // 执行，会按同一份旧余额重复开仓。60s 僵死兜底：异常挂起时不永久卡死主循环。
   397→  private ticking = false;
   398→  private tickStartedAt = 0;
   399→
   400→  async tick(): Promise<void> {
   401→    if (this.stopped) return;
   402→    if (this.ticking && Date.now() - this.tickStartedAt < 60_000) return;
   403→    this.ticking = true;
   404→    this.tickStartedAt = Date.now();
   405→    try {
   406→      const now = Date.now();
   407→      this.lastTickAt = now;
   408→      const cfg = await this.repo.loadConfig(TREND_TRADE.ACCOUNT_NAME);
   409→      if (!cfg) return;
   410→      const acc = await db.paperAccount.findUnique({ where: { name: TREND_TRADE.ACCOUNT_NAME } });
   411→      if (!acc) return;
   412→      const accId = acc.id;
   413→
   414→      // ① 持续风险巡检（任何模式都执行——强平不因暂停/数据不可用而缺席）
   415→      await this.engine.liquidationCheck(accId);
   416→      await this.finalizeNewLiquidations();
   417→
   418→      // ② 持仓结果采集（§七：MFE/MAE 按当前可用标记价增量更新——任何模式都执行）
   419→      await this.updateOpenStats();
   420→
   421→      // ②b 策略自愈：DB 配置的算法与实例不一致（如 control 已写库但实例切换失败）→ 重新装配
   422→      if (!this.fixedStrategy && cfg.strategyId && cfg.strategyId !== this.strategy.id) {
   423→        await this.applyStrategy(cfg.strategyId);
   424→        logDecision(`策略自愈切换 → ${cfg.strategyId}`);
   425→      }
   426→
   427→      // ③ 退出管理（暂停模式也执行：趋势结束/反向确认/策略离场照常执行）
   428→      const signals = this.deps.getSignals();
   429→      const bySymbol = new Map(signals.map((s) => [s.symbol, s]));
   430→      await this.manageExits(bySymbol);
   431→
   432→      // ③b 策略行情采样（冲高检测等需要短窗价格序列；只喂当时可用的标记价）
   433→      for (const s of signals) {
   434→        const mark = this.deps.markPrice(s.symbol);
   435→        if (mark > 0) this.strategy.observe(s.symbol, mark, now);
   436→      }
   437→      for (const meta of await this.repo.loadMetasByStatus('OPEN')) {
   438→        const mark = this.deps.markPrice(meta.symbol);
   439→        if (mark > 0) this.strategy.observe(meta.symbol, mark, now);
   440→      }
   441→
   442→      // ④ 权益峰值（回撤基准；任何模式都更新）
   443→      const eq = await this.accountEquity(accId);
   444→      if (eq.equity > cfg.peakEquity) {
   445→        await this.repo.updatePeakEquity(TREND_TRADE.ACCOUNT_NAME, eq.equity);
   446→      }
   447→
   448→      // ⑤ 开仓（仅 RUNNING 且数据新鲜；暂停/停止只停新开仓与待入场转换）
   449→      if (cfg.mode === 'RUNNING') {
   450→        await this.managePendings(bySymbol, cfg, eq); // 先转换已到期的待入场（等一根 K 线复核）
   451→        await this.manageEntries(cfg, signals, eq); // 再为策略想开仓的趋势创建待入场记录
   452→      }
   453→      this.lastError = null;
   454→    } catch (e) {
   455→      this.lastError = e instanceof Error ? e.message : String(e);
   456→      console.warn('[trend-auto] tick 失败：', this.lastError);
   457→    } finally {
   458→      this.ticking = false;
   459→    }
   460→  }
   461→
   462→  // ============ 持仓结果采集（Task 56 §七） ============
   463→
   464→  /**
   465→   * MFE/MAE 增量更新：只用当前 tick 可得的标记价（多源供价的当前快照），逐 tick 维护
   466→   * 「持仓期间」的最大有利/不利浮动盈亏（占初始保证金 %）——绝不使用事后才知道的
   467→   * 最高/最低价构造当时不存在的数据。MFE 更新时刻与峰值后回撤一并落库。
   468→   */
   469→  private async updateOpenStats(): Promise<void> {
   470→    const openMetas = await this.repo.loadMetasByStatus('OPEN');
   471→    for (const meta of openMetas) {
   472→      try {
   473→        const pos = await db.paperPosition.findUnique({ where: { id: meta.positionId } });
   474→        if (!pos || pos.status !== 'OPEN' || !(pos.margin > 0)) continue;
   475→        const mark = this.deps.markPrice(meta.symbol);
   476→        if (!(mark > 0)) continue; // 标记价不可用：跳过本 tick（不用过期价格虚构统计）
   477→        const sign = meta.side === 'LONG' ? 1 : -1;
   478→        const pnlPct = (((mark - pos.entryPrice) * pos.qty * sign) / pos.margin) * 100;
   479→        const prevMfe = meta.mfe ?? null;
   480→        const mfe = prevMfe != null ? Math.max(prevMfe, pnlPct) : pnlPct;
   481→        const mae = meta.mae != null ? Math.min(meta.mae, pnlPct) : pnlPct;
   482→        const isNewPeak = prevMfe == null || pnlPct > prevMfe;
   483→        const mfeAt = isNewPeak ? Date.now() : (meta.mfeAt ?? null);
   484→        const mfeDrawdown = mfe - pnlPct;
   485→        if (
   486→          prevMfe == null ||
   487→          meta.mae == null ||
   488→          Math.abs(mfe - prevMfe) > 1e-9 ||
   489→          Math.abs(mae - meta.mae) > 1e-9
   490→        ) {
   491→          await this.repo.updateTradeStats(meta.positionId, { mfe, mae, mfeAt, mfeDrawdown });
   492→        }
   493→      } catch {
   494→        // 单笔统计失败不影响主循环
   495→      }
   496→    }
   497→  }
   498→
   499→  // ============ 退出管理 ============
   500→
   501→  /**
   502→   * 退出管理 v4（Task 57：基础设施硬规则优先，策略离场在其内表达）：
   503→   * ① ENDED（lifecycle 已按「连续 N 根不同已收盘 K 线持续证据」确认）→ 平仓，但必须：
   504→   *    a) 信号携带的 trendCycleId 与持仓绑定周期一致（旧周期 ENDED 不杀新周期持仓，
   505→   *       也不仅凭 symbol 平仓）；
   506→   *    b) 信号最近 K 线晚于开仓依据的 K 线（开仓所在 K 线不得作为该持仓的结束证据）。
   507→   * ② END_PENDING（同周期出现结束迹象）→ 只记录首次报警时刻（firstEndSignAt）；
   508→   *    是否提前离场交给策略（短周期套利：走弱先走，走强后按轮次再入场）。
   509→   * ③ 策略离场（evaluateExit）：SPIKE_TAKE 冲高套利 / WEAKNESS_EARLY 走弱提前离场——
   510→   *    基于当时可用的标记价短窗与信号相位，不使用未来数据。
   511→   * ④ 反向趋势已确认（CONFIRMED_*，方向与持仓相反，dataFresh）→ 立即平仓。
   512→   * 数据缺失（DATA_UNAVAILABLE）永不触发离场；信号缺失（退池）不误平仓，仅强平/人工管理。
   513→   */
   514→  private async manageExits(bySymbol: Map<string, UnifiedTrendSignal>): Promise<void> {
   515→    const openMetas = await this.repo.loadMetasByStatus('OPEN');
   516→    for (const meta of openMetas) {
   517→      const signal = bySymbol.get(meta.symbol);
   518→      if (!signal) continue; // 信号缺失（标的退池/未评估）：不误平仓，仅强平/人工管理
   519→      const favorable = meta.side === 'LONG' ? signal.direction === 'up' : signal.direction === 'down';
   520→
   521→      // ① 趋势已结束（ENDED）→ 周期匹配 + 证据时序校验后自行平仓
   522→      if (signal.phase === 'ENDED') {
   523→        if (!signal.trendCycleId || signal.trendCycleId !== meta.trendCycleId) {
   524→          // 旧周期 ENDED 归属他人——不得据此平掉本周期持仓
   525→          logDecision(
   526→            `离场跳过 ENDED 周期不匹配 symbol=${meta.symbol} posCycle=${meta.trendCycleId} ` +
   527→              `signalCycle=${signal.trendCycleId ?? '-'} positionId=${meta.positionId}`,
   528→          );
   529→          continue;
   530→        }
   531→        const entryBar = this.entryBarTimeOf(meta);
   532→        if (entryBar != null && signal.lastBarTime != null && signal.lastBarTime <= entryBar) {
   533→          // 结束证据未出现「开仓之后的新收盘 K 线」——不得平仓
   534→          logDecision(
   535→            `离场跳过 证据K线未晚于开仓K线 symbol=${meta.symbol} posCycle=${meta.trendCycleId} ` +
   536→              `entryBar=${entryBar} signalBar=${signal.lastBarTime} positionId=${meta.positionId}`,
   537→          );
   538→          continue;
   539→        }
   540→        await this.closePosition(meta, 'TREND_ENDED', signal);
   541→        continue;
   542→      }
   543→
   544→      // ② 本周期出现结束迹象（模型已报警）→ 只记录首次报警时刻；是否提前离场由策略决定
   545→      if (signal.phase === 'END_PENDING' && signal.trendCycleId === meta.trendCycleId) {
   546→        if (signal.endPendingSince != null && meta.firstEndSignAt == null) {
   547→          await this.repo.markFirstEndSign(meta.positionId, signal.endPendingSince);
   548→          logDecision(
   549→            `结束迹象首次报警（是否提前离场由策略「${this.strategy.name}」决定） ` +
   550→              `symbol=${meta.symbol} posCycle=${meta.trendCycleId} ev=${signal.endEvidenceBars} ` +
   551→              `positionId=${meta.positionId}`,
   552→          );
   553→        }
   554→      }
   555→
   556→      // ③ 策略离场（SPIKE_TAKE 冲高套利 / WEAKNESS_EARLY 走弱先走；基于当时可用数据）
   557→      const mark = this.deps.markPrice(meta.symbol);
   558→      if (mark > 0) {
   559→        const pos = await db.paperPosition.findUnique({ where: { id: meta.positionId } });
   560→        if (pos && pos.status === 'OPEN') {
   561→          const v = this.strategy.evaluateExit({
   562→            meta,
   563→            pos: {
   564→              entryPrice: pos.entryPrice,
   565→              qty: pos.qty,
   566→              margin: pos.margin,
   567→              side: meta.side as 'LONG' | 'SHORT',
   568→              openedAt: pos.openedAt instanceof Date ? pos.openedAt.getTime() : Number(pos.openedAt),
   569→            },
   570→            mark,
   571→            signal,
   572→            now: Date.now(),
   573→          });
   574→          if (v.exit && v.reason) {
   575→            logDecision(`策略离场 [${this.strategy.id}] ${v.detail} positionId=${meta.positionId}`);
   576→            await this.closePosition(meta, v.reason, signal);
   577→            continue;
   578→          }
   579→        }
   580→      }
   581→
   582→      // ④ 反向趋势已确认（独立确认 + 数据新鲜）→ 自行平仓（反手开仓见 manageEntries 守卫）
   583→      if (
   584→        (signal.phase === 'CONFIRMED_UP' || signal.phase === 'CONFIRMED_DOWN') &&
   585→        !favorable &&
   586→        signal.dataFresh
   587→      ) {
   588→        await this.closePosition(meta, 'REVERSE_CONFIRMED', signal);
   589→      }
   590→    }
   591→  }

   593→  /** 开仓依据的已收盘 K 线（§2.3 守卫用；旧数据缺列时回退到 signalAtOpen.lastBarTime） */
   594→  private entryBarTimeOf(meta: TrendTradeMetaRow): number | null {
   595→    if (meta.entryBarTime != null) return meta.entryBarTime;
   596→    try {
   597→      const s = JSON.parse(meta.signalAtOpen) as { lastBarTime?: number | null };
   598→      return typeof s.lastBarTime === 'number' ? s.lastBarTime : null;
   599→    } catch {
   600→      return null;
   601→    }
   602→  }

   604→  /** 平仓（幂等护栏 = 仓位状态本身；决策行事后审计记录） */
   605→  private async closePosition(
   606→    metaIn: TrendTradeMetaRow,
   607→    reason: TrendExitReason,
   608→    signal: UnifiedTrendSignal | null,
   609→  ): Promise<void> {
   610→    try {
   611→      // 重读最新 meta（同轮 tick 内 ② 可能已写入 firstEndSignAt——避免旧快照覆盖回 null）
   612→      const meta = (await this.repo.getMeta(metaIn.positionId)) ?? metaIn;
   613→      const pos = await db.paperPosition.findUnique({ where: { id: meta.positionId } });
   614→      if (!pos || pos.status !== 'OPEN') {
   615→        // 已终态（并发/重启窗口）：补写 meta 终态
   616→        await this.repo.upsertMeta({
   617→          ...meta,
   618→          status: 'CLOSED',
   619→          exitReason: pos?.status === 'LIQUIDATED' ? 'LIQUIDATED' : reason,
   620→          netPnlWithSlip: (pos?.pnl ?? 0) - meta.slippageEst,
   621→          closedAt: pos?.closedAt ?? new Date(),
   622→        });
   623→        return;
   624→      }
   625→      const r = await this.engine.close(meta.positionId);
   626→      const mark = r.price;
   627→      const closeSlip = mark * pos.qty * (TREND_TRADE.SLIPPAGE_BPS / 10_000);
   628→      const netWithSlip = r.pnl - meta.slippageEst - closeSlip;
   629→      await this.repo.upsertMeta({
   630→        ...meta,
   631→        status: 'CLOSED',
   632→        exitReason: reason,
   633→        exitSignalJson: signal ? JSON.stringify(signal) : null,
   634→        netPnlWithSlip: netWithSlip,
   635→        closedAt: new Date(),
   636→        exitEndReason: signal?.endReason ?? meta.exitEndReason ?? null,
   637→        exitEvidenceBars: signal?.endEvidenceBars ?? meta.exitEvidenceBars ?? null,
   638→        closeNotional: mark * pos.qty,
   639→      });
   640→      const round = meta.entryRound ?? 1;
   641→      await this.repo.claimDecision({
   642→        id: `${roundKey(meta.trendCycleId, round)}:CLOSE`,
   643→        symbol: meta.symbol,
   644→        action: reason === 'REVERSE_CONFIRMED' ? 'REVERSE_CLOSE' : 'CLOSE',
   645→        cycleId: meta.trendCycleId,
   646→        positionId: meta.positionId,
   647→        detail: JSON.stringify({
   648→          reason,
   649→          round,
   650→          strategyId: this.strategy.id,
   651→          exitEndReason: signal?.endReason ?? null,
   652→          exitEvidenceBars: signal?.endEvidenceBars ?? null,
   653→          exitPrice: r.price,
   654→          netPnl: r.pnl,
   655→          netPnlWithSlip: netWithSlip,
   656→          mfe: meta.mfe ?? null,
   657→          mae: meta.mae ?? null,
   658→          endReason: signal?.endReason ?? null,
   659→        }),
   660→      });
   661→      // 策略回调：保留轮次状态供「走强再入场」判定
   662→      this.strategy.onClose(meta.symbol, meta.trendCycleId, reason, Date.now());
   663→      this.closes++;
   664→      logDecision(
   665→        `平仓 reason=${reason} strategy=${this.strategy.id} round=${round} endReason=${signal?.endReason ?? '-'} symbol=${meta.symbol} ` +
   666→          `side=${meta.side} posCycle=${meta.trendCycleId} positionId=${meta.positionId} ` +
   667→          `ev=${signal?.endEvidenceBars ?? '-'} exitPrice=${mark} netPnl=${r.pnl.toFixed(4)} ` +
   668→          `netWithSlip=${netWithSlip.toFixed(4)} mfe=${meta.mfe?.toFixed(1) ?? '-'}% mae=${meta.mae?.toFixed(1) ?? '-'}%`,
   669→      );
   670→    } catch (e) {
   671→      // 平仓暂时性失败（行情不可用等）：下一轮重试（决策行未写，不烧掉本周期）
   672→      // ⚠️ 此处只能用入参 metaIn——重读的 meta 声明在 try 作用域内，catch 引用会 ReferenceError
   673→      if (!(e instanceof PaperError)) {
   674→        this.pushSkip(metaIn.symbol, `平仓异常：${e instanceof Error ? e.message : e}`, metaIn.trendCycleId);
   675→      }
   676→    }
   677→  }

   679→  /** 强平巡检后的 meta 终态补写（强平是执行结果，如实记录、不伪装模型止损） */
   680→  private async finalizeNewLiquidations(): Promise<void> {
   681→    const openMetas = await this.repo.loadMetasByStatus('OPEN');
   682→    for (const m of openMetas) {
   683→      const pos = await db.paperPosition.findUnique({ where: { id: m.positionId } });
   684→      if (pos && pos.status === 'LIQUIDATED') {
   685→        await this.finalizeLiquidated(m, pos.margin, pos.pnl ?? -pos.margin);
   686→      }
   687→    }
   688→  }
   689→
   690→  private async finalizeLiquidated(m: TrendTradeMetaRow, margin: number, pnl: number): Promise<void> {
   691→    await this.repo.upsertMeta({
   692→      ...m,
   693→      status: 'CLOSED',
   694→      exitReason: 'LIQUIDATED',
   695→      netPnlWithSlip: pnl - m.slippageEst,
   696→      closedAt: new Date(),
   697→    });
   698→    const round = m.entryRound ?? 1;
   699→    await this.repo.claimDecision({
   700→      id: `${roundKey(m.trendCycleId, round)}:CLOSE`,
   701→      symbol: m.symbol,
   702→      action: 'CLOSE',
   703→      cycleId: m.trendCycleId,
   704→      positionId: m.positionId,

   708→    this.closes++;
   709→    logDecision(
   710→      `强平 symbol=${m.symbol} side=${m.side} posCycle=${m.trendCycleId} round=${round} positionId=${m.positionId} ` +

   715→  // ============ 开仓决策（Task 57：方向/门槛/模式判定全部交给策略） ============

   717→  /** 策略入场评估上下文（快照；策略不得改动） */
   718→  private strategyCtx(
   719→    now: number,
   720→    mark: number,
   721→    cfg: TrendAutoConfigRow,
   722→    eq: { balance: number; marginUsed: number; unrealized: number; equity: number; openCount: number },
   723→    openSymbols: Set<string>,

   729→  /** 开仓时高周期门控状态标签（快照存档用；方向分级与策略同源） */
   730→  private htfGateLabel(signal: UnifiedTrendSignal): HtfGateState {
   731→    const trendDir = signal.direction === 'down' ? 'down' : 'up';
   732→    const g5 = classifyHtf(signal.htf5m.dir, signal.htf5m.valid, trendDir);
   733→    const g15 = classifyHtf(signal.htf15m.dir, signal.htf15m.valid, trendDir);
   734→    if (g5 === 'INVALID' || g15 === 'INVALID') return 'INVALID';
   735→    if (g5 === 'OPPOSED') return 'OPPOSED';
   736→    if (g5 === 'NEUTRAL') return g15 === 'OPPOSED' ? 'OPPOSED' : 'NEUTRAL';
   737→    return g15; // 5m ALIGNED：标签反映 15m 状态（ALIGNED/NEUTRAL/OPPOSED）
   738→  }
   739→
   740→  // ============ 待入场转换 / 新信号扫描（Task 57：门槛判定全部由当前策略执行） ============
   741→
   742→  /**
   743→   * 待入场管理（二次确认）：每个 ACTIVE 待入场记录在出现「更新的已收盘 K 线」后
   744→   * 全量复核（策略门槛/相位/周期/数据/冷却/容量/资金），通过才开仓；趋势失效、数据过期、
   745→   * 门槛不再满足或记录过期 → 取消并记录原因。每轮次至多一条记录、只允许成功开仓一次
   746→   * （决策键 `{cycleId}#R{round}:OPEN` 数据库唯一约束兜底）。
   747→   */
   748→  private async managePendings(
   749→    bySymbol: Map<string, UnifiedTrendSignal>,
   750→    cfg: TrendAutoConfigRow,
   751→    eq: { balance: number; marginUsed: number; unrealized: number; equity: number; openCount: number },
   752→  ): Promise<void> {
   753→    const pendings = await this.repo.loadActivePendings();
   754→    for (const p of pendings) {
   755→      const signal = bySymbol.get(p.symbol);
   756→      const cancel = async (reason: string): Promise<void> => {
   757→        await this.repo.markPendingCanceled(p.cycleId, reason);
   758→        logDecision(
   759→          `待入场取消 roundKey=${p.cycleId} symbol=${p.symbol} reason=${reason} ` +
   760→            `ageMs=${Date.now() - p.firstSeenAt}`,
   761→        );
   762→      };
   763→      // ① 记录过期（等待超时——趋势确认后长期未出现合格的新 K 线）
   764→      if (Date.now() - p.firstSeenAt > TREND_TRADE.PENDING_EXPIRE_MS) {
   765→        await cancel('待入场超时（等待新收盘 K 线复核超限）');
   766→        continue;
   767→      }
   768→      // ② 信号缺失/数据过期/相位失效/周期变化 → 取消（§5.3：趋势失效或数据过期取消记录）
   769→      if (!signal) {
   770→        await cancel('信号缺失（标的退池/未评估）');
   771→        continue;
   772→      }
   773→      if (!signal.dataFresh) {
   774→        await cancel(`数据过期（${signal.dataUnavailableReason ?? '未知'}）`);
   775→        continue;
   776→      }
   777→      if (
   778→        !(TREND_TRADE.CONFIRMED_PHASES as readonly string[]).includes(signal.phase) ||
   779→        signal.trendCycleId !== baseCycleOf(p.cycleId)
   780→      ) {
   781→        await cancel(`趋势失效（phase=${signal.phase}，cycle=${signal.trendCycleId ?? '-'}）`);
   782→        continue;
   783→      }
   784→      // ③ 还没等到「新的已收盘 K 线」→ 继续等待（二次确认核心等待条件）
   785→      if (signal.lastBarTime == null || signal.lastBarTime <= p.firstBarTime) continue;
   786→      // ④ 冷却复核：冷却以 symbol 为单位，不因周期/轮次变化绕过（时长控制台可调）
   787→      const lastClose = await this.repo.lastCloseAt(p.symbol);
   788→      if (lastClose != null && Date.now() - lastClose < cooldownOf(cfg)) {
   789→        await cancel('转换时进入同币种冷却期');
   790→        continue;
   791→      }
   792→      // ⑤ 策略门槛全量复核（方向/模式/评分/ER/轮次——与创建时同一策略判定）
   793→      const ctx = this.strategyCtx(
   794→        Date.now(),
   795→        this.deps.markPrice(p.symbol),
   796→        cfg,
   797→        eq,
   798→        new Set(),
   799→        lastClose,
   800→      );
   801→      const verdict = this.strategy.evaluateEntry(signal, ctx);
   802→      if (!verdict.want) {
   803→        await cancel(`复核未过门槛：${verdict.reason}`);
   804→        continue;
   805→      }
   806→      // ⑥-⑩ 开仓（共享 executeOpen：容量/同币防御/资金测算/幂等抢占/开仓/审计全同构——
   807→      // 待入场转换与「二次确认关闭→直接开仓」两条路径走同一实现，行为不分叉）
   808→      const outcome = await this.executeOpen({
   809→        signal,
   810→        rKey: p.cycleId,
   811→        side: p.side as 'LONG' | 'SHORT',
   812→        verdict,
   813→        cfg,
   814→        eq,
   815→        via: 'pending',
   816→        waitedSince: p.firstSeenAt,
   817→        onConverted: (positionId) => this.repo.markPendingConverted(p.cycleId, positionId),
   818→      });
   819→      if (outcome.outcome === 'opened') continue;
   820→      if (outcome.outcome === 'capacity') continue; // 保留记录，等下轮复核直到过期（审计已在 executeOpen 记录）
   821→      if (outcome.outcome === 'transient') {
   822→        // 行情暂时不可用：释放待入场记录，下轮同周期可重建重试（决策已被 executeOpen 释放）
   823→        await this.repo.deletePending(p.cycleId);
   824→        continue;
   825→      }
   826→      // skipped：资金/上限/防御类永久跳过——取消待入场并记录原因（审计行已在 executeOpen 落库）
   827→      await cancel(outcome.reason ?? '开仓跳过');
   828→    }
   829→  }
   830→
   831→  /**
   832→   * 尝试把「策略想开仓」落成真实持仓（Task 57-c 抽取：待入场转换 / 直接开仓共用）。
   833→   * 守卫顺序与 Task 56 验收语义一致：容量 → 同币防御 → 资金测算 → 幂等抢占 → 开仓 →
   834→   * meta/审计/回调。返回结果由调用方决定待入场记录的保留/取消/释放。
   835→   */
   836→  private async executeOpen(input: {
   837→    signal: UnifiedTrendSignal;
   838→    /** `{cycleId}#R{round}`（幂等键与轮次的唯一定位） */
   839→    rKey: string;
   840→    side: 'LONG' | 'SHORT';
   841→    verdict: StrategyEntryVerdict;
   842→    cfg: TrendAutoConfigRow;
   843→    eq: { balance: number; marginUsed: number; unrealized: number; equity: number; openCount: number };
   844→    /** 审计来源：pending = 待入场转换（二次确认通过）/ direct = 直接开仓（二次确认关闭） */
   845→    via: 'pending' | 'direct';
   846→    /** 等待起点（pending = firstSeenAt；direct 无）——审计 waitedMs 用 */
   847→    waitedSince?: number;
   848→    /** pending 专属：开仓成功后把待入场记录标记 CONVERTED */
   849→    onConverted?: (positionId: string) => Promise<void>;
   850→  }): Promise<
   851→    | { outcome: 'opened' }
   852→    | { outcome: 'capacity' }
   853→    | { outcome: 'skipped'; reason: string }
   854→    | { outcome: 'transient'; reason: string }
   855→  > {
   856→    const { signal, rKey, side, verdict, cfg, eq, via } = input;
   857→    const posCycle = baseCycleOf(rKey);
   858→    const skipLabel = via === 'pending' ? '待入场转换' : '直接开仓';
   859→
   860→    // ⑥ 容量守卫：pending 保留记录继续等；direct 仅记录跳过
   861→    if (eq.openCount >= cfg.maxPositions) {
   862→      this.pushSkip(
   863→        signal.symbol,
   864→        `${skipLabel}等待容量（已达最大持仓数 ${cfg.maxPositions}）`,
   865→        rKey,
   866→      );
   867→      return { outcome: 'capacity' };
   868→    }
   869→    // ⑦ 同 symbol 已有持仓（防御；正常由周期决策键保证）
   870→    const dup = await db.paperPosition.findFirst({
   871→      where: { accountId: await this.autoAccountId(), symbol: signal.symbol, status: 'OPEN' },
   872→    });
   873→    if (dup) {
   874→      return { outcome: 'skipped', reason: '同币种已有持仓' };
   875→    }
   876→    // ⑧ 仓位测算（Task 58：每仓保证金 = 权益 × min(设定比例, 95%÷持仓数)——持仓数多时
   877→    //    自动压缩单仓比例保证满仓可开；超出可用余额直接跳过）。费率唯一真值 = 引擎实例
   878→    //    feeRate（Task 59：自动账户 0.02%；测试注入引擎默认 TAKER_FEE，口径随引擎走）
   879→    const feeRate = this.engine.feeRate;
   880→    const feeFactor = 1 + cfg.leverage * feeRate;
   881→    const effRatio = effectiveMarginRatio(cfg.marginRatio, cfg.maxPositions);
   882→    const effRatioPct = (effRatio * 100).toFixed(2).replace(/\.?0+$/, '');
   883→    const targetMargin = floor2(eq.equity * effRatio);
   884→    const fee = targetMargin * cfg.leverage * feeRate;
   885→    if (targetMargin + fee > eq.balance) {
   886→      const reason = `权益 ${effRatioPct}% 保证金 ${targetMargin.toFixed(2)} + 手续费 ${fee.toFixed(2)} 超出可用余额 ${eq.balance.toFixed(2)}（跳过不变通）`;
   887→      await this.recordSkipDb(signal.symbol, rKey, reason);
   888→      return { outcome: 'skipped', reason };
   889→    }
   890→    if (targetMargin < TREND_TRADE.MIN_MARGIN) {
   891→      const reason = `权益 ${effRatioPct}% 保证金 ${targetMargin} 低于最小保证金 ${TREND_TRADE.MIN_MARGIN}`;
   892→      await this.recordSkipDb(signal.symbol, rKey, reason);
   893→      return { outcome: 'skipped', reason };
   894→    }
   895→    // ⑨ 幂等守卫（数据库唯一约束兜底）：抢占 `{cycleId}#R{round}:OPEN` 决策
   896→    const claimed = await this.repo.claimDecision({
   897→      id: `${rKey}:OPEN`,
   898→      symbol: signal.symbol,
   899→      action: 'OPEN',
   900→      cycleId: posCycle,
   901→      detail: JSON.stringify(this.openDetail(signal, targetMargin, cfg, verdict.modeTag)),
   902→    });
   903→    if (!claimed) {
   904→      return { outcome: 'skipped', reason: '本趋势周期本轮次已决策过（幂等）' };
   905→    }
   906→    try {
   907→      const pos = await this.engine.open({
   908→        symbol: signal.symbol,
   909→        side,
   910→        leverage: cfg.leverage,
   911→        margin: targetMargin,
   912→      });
   913→      const slippageEst = targetMargin * cfg.leverage * (cfg.slippageBps / 10_000);
   914→      await this.repo.upsertMeta({
   915→        positionId: pos.id,
   916→        trendCycleId: posCycle,
   917→        symbol: signal.symbol,
   918→        side,
   919→        trendPhaseAtOpen: signal.phase,
   920→        scoreAtOpen: signal.score,
   921→        featuresAtOpen: JSON.stringify({
   922→          strategyId: this.strategy.id,
   923→          modeTag: verdict.modeTag,
   924→          vol: signal.features,
   925→          htf5m: signal.htf5m,
   926→          htf15m: signal.htf15m,
   927→          htfAgree: signal.htfAgree,
   928→          strength: signal.strength,
   929→          anomaly: signal.anomaly,
   930→          conflict: signal.conflict,
   931→        }),
   932→        signalAtOpen: JSON.stringify(signal),
   933→        slippageEst,
   934→        status: 'OPEN',
   935→        exitReason: null,
   936→        exitSignalJson: null,
   937→        netPnlWithSlip: null,
   938→        openDecisionAt: new Date(),
   939→        closedAt: null,
   940→        // 开仓快照（可查询列）
   941→        entryEr: signal.features.efficiencyRatio,
   942→        htf5mDir: signal.htf5m.dir,
   943→        htf15mDir: signal.htf15m.dir,
   944→        htfGate: this.htfGateLabel(signal),
   945→        entryNotional: targetMargin * cfg.leverage,
   946→        entryBarTime: signal.lastBarTime,
   947→        mfe: null,
   948→        mae: null,
   949→        mfeAt: null,
   950→        mfeDrawdown: null,
   951→        firstEndSignAt: null,
   952→        exitEndReason: null,
   953→        exitEvidenceBars: null,
   954→        closeNotional: null,
   955→        entryRound: pendingRoundOf(rKey),
   956→      });
   957→      if (input.onConverted) await input.onConverted(pos.id);
   958→      await this.attachPosition(`${rKey}:OPEN`, pos.id);
   959→      // 策略回调：轮次 +1 / 新周期初始化（持久化到 TrendStrategyState）
   960→      this.strategy.onOpen(signal.symbol, posCycle, signal.lastBarTime, Date.now());
   961→      this.opens++;
   962→      eq.openCount += 1;
   963→      eq.marginUsed += targetMargin;
   964→      eq.balance -= targetMargin + fee;
   965→      eq.equity -= fee; // equity = balance + marginUsed + unrealized 自洽（开仓瞬时浮亏≈0）
   966→      const waitedMs =
   967→        input.waitedSince != null ? ` waitedMs=${Date.now() - input.waitedSince}` : '';
   968→      logDecision(
   969→        `开仓（${via === 'pending' ? '二次确认通过' : '直接开仓·二次确认关闭'}）strategy=${this.strategy.id} mode=${verdict.modeTag} ` +
   970→          `round=${pendingRoundOf(rKey)} symbol=${signal.symbol} side=${side} posCycle=${posCycle} ` +
   971→          `positionId=${pos.id} margin=${targetMargin}(${effRatioPct}%权益=min(${Math.round(cfg.marginRatio * 100)}%,95%/${cfg.maxPositions}仓)) ` +
   972→          `leverage=${cfg.leverage}x entryBar=${signal.lastBarTime}${waitedMs} ` +
   973→          `score=${signal.score.toFixed(0)} er=${signal.features.efficiencyRatio.toFixed(2)} ` +
   974→          `htf5m=${signal.htf5m.dir} htf15m=${signal.htf15m.dir}`,
   975→      );
   976→      return { outcome: 'opened' };
   977→    } catch (e) {
   978→      if (isTransientOpenError(e)) {
   979→        // 行情暂时不可用：释放决策（下轮同周期可重建/重试）
   980→        await this.releaseDecision(`${rKey}:OPEN`);
   981→        this.pushSkip(signal.symbol, `开仓暂时失败（将重试）：${e instanceof Error ? e.message : e}`, rKey);
   982→        return { outcome: 'transient', reason: e instanceof Error ? e.message : String(e) };
   983→      }
   984→      // 永久跳过（保证金/上限类）：决策保留防重入
   985→      const msg = e instanceof Error ? e.message : String(e);
   986→      this.pushSkip(signal.symbol, `开仓跳过：${msg}`, rKey);
   987→      return { outcome: 'skipped', reason: `开仓跳过：${msg}` };
   988→    }
   989→  }
   990→
   991→  /**
   992→   * 新信号扫描（Task 57）：对每个「数据新鲜 + 确认相位」的趋势信号询问当前策略是否想开仓
   993→   * （方向/模式/评分/ER/轮次上限全部由策略判定）。
   994→   * Task 57-c 二次确认开关（waitBarConfirm，控制台可调）：
   995→   * - 开（默认）：创建待入场记录（主键 `{cycleId}#R{round}`，等下一根已收盘 K 线复核），不立即下单；
   996→   * - 关：同 tick 直接开仓（executeOpen 全量守卫：容量/同币/资金/幂等）——快 1 根 K 线，少一层防护。
   997→   * 同周期轮次上限、同周期再入场间隔与新 K 线要求由策略内部约束。
   998→   */
   999→  private async manageEntries(
  1000→    cfg: TrendAutoConfigRow,
  1001→    signals: UnifiedTrendSignal[],
  1002→    eq: { balance: number; marginUsed: number; unrealized: number; equity: number; openCount: number },
  1003→  ): Promise<void> {
  1004→    // 容量核算：已开仓 + 待入场（其 symbol 无持仓者，复核后将各占一仓）共同计入上限——
  1005→    // 防止待入场批量创建超过 maxPositions（转换层另有二次守卫）
  1006→    const accountId = await this.autoAccountId();
  1007→    const openRows = await db.paperPosition.findMany({ where: { accountId, status: 'OPEN' } });
  1008→    const openSymbols = new Set(openRows.map((p) => p.symbol));
  1009→    const activePendings = await this.repo.loadActivePendings();
  1010→    const pendingSymbols = new Set(
  1011→      activePendings.filter((p) => !openSymbols.has(p.symbol)).map((p) => p.symbol),
  1012→    );
  1013→
  1014→    for (const signal of signals) {
  1015→      // 非确认相位静默跳过（策略同样判定；此处仅为避免审计日志被「未确认」刷屏）
  1016→      if (!(TREND_TRADE.CONFIRMED_PHASES as readonly string[]).includes(signal.phase)) continue;
  1017→      if (!signal.dataFresh) {
  1018→        this.pushSkip(signal.symbol, '数据不可用/过期（不开新仓）', signal.trendCycleId ?? undefined);
  1019→        continue;
  1020→      }
  1021→      const cycleId = signal.trendCycleId;
  1022→      if (!cycleId) continue;
  1023→
  1024→      // 幂等守卫 1：同 symbol 已有 OPEN 关联持仓（防御；正常由周期决策键保证）
  1025→      if (openSymbols.has(signal.symbol)) continue;
  1026→
  1027→      // 本 symbol 已存在 ACTIVE 待入场记录（含历史轮次键）：不重复创建
  1028→      const activePending = await this.repo.getActivePendingBySymbol(signal.symbol);
  1029→      if (activePending) continue;
  1030→
  1031→      // 本轮次决策已抢占过（幂等键烧毁保护——转换层取消的场景由 pending 重建覆盖）
  1032→      const round = this.strategy.nextRound(signal.symbol, cycleId);
  1033→      const rKey = roundKey(cycleId, round);
  1034→      const prevRound = await this.repo.getPendingByCycle(rKey);
  1035→      if (prevRound) {
  1036→        if (prevRound.status === 'ACTIVE' || prevRound.status === 'CONVERTED') continue;
  1037→        // CANCELED：删除后同轮次可重建（容量/冷却等暂时性取消不烧轮次；审计留决策行）
  1038→        await this.repo.deletePending(rKey);
  1039→      }
  1040→
  1041→      // 容量守卫（跳过不烧轮次：仓位/待入场释放后同一确认趋势仍可创建待入场）
  1042→      if (eq.openCount + pendingSymbols.size >= cfg.maxPositions) {
  1043→        this.recordSkip(signal.symbol, `已达最大持仓数 ${cfg.maxPositions}`, rKey);
  1044→        continue;
  1045→      }
  1046→
  1047→      // 同币种冷却：平仓后冷却期内新信号只记录不执行（时长控制台可调）；到期后重走完整门槛
  1048→      const lastClose = await this.repo.lastCloseAt(signal.symbol);
  1049→      if (lastClose != null && Date.now() - lastClose < cooldownOf(cfg)) {
  1050→        this.recordSkip(
  1051→          signal.symbol,
  1052→          `同币种冷却中（距上次平仓 ${Math.ceil((Date.now() - lastClose) / 1000)}s / ${cooldownOf(cfg) / 1000}s）`,
  1053→          rKey,
  1054→        );
  1055→        continue;
  1056→      }
  1057→
  1058→      // 「先平后开」独立性：确认评估包含的已收盘 K 线必须晚于该 symbol 上一笔平仓
  1059→      // （同一根 K 线不既触发平仓又触发反向开仓；首笔无平仓记录直接放行）
  1060→      if (TREND_TRADE.REVERSE_INDEPENDENT_BARS && lastClose != null) {
  1061→        if (signal.lastBarTime != null && signal.lastBarTime + 60_000 <= lastClose) {
  1062→          this.recordSkip(signal.symbol, '等待平仓后下一根已收盘 K 线确认（先平后开）', rKey);
  1063→          continue;
  1064→        }
  1065→      }
  1066→
  1067→      // 策略入场评估（方向/模式/门槛/轮次——唯一决策来源）
  1068→      const mark = this.deps.markPrice(signal.symbol);
  1069→      if (!(mark > 0)) {
  1070→        this.pushSkip(signal.symbol, '标记价不可用（跳过本 tick）', rKey);
  1071→        continue;
  1072→      }
  1073→      const ctx = this.strategyCtx(Date.now(), mark, cfg, eq, openSymbols, lastClose);
  1074→      const verdict = this.strategy.evaluateEntry(signal, ctx);
  1075→      if (!verdict.want) {
  1076→        this.recordSkip(signal.symbol, `策略不入场：${verdict.reason}`, rKey);
  1077→        continue;
  1078→      }
  1079→      const side = verdict.side ?? (signal.phase === 'CONFIRMED_UP' ? 'LONG' : 'SHORT');
  1080→
  1081→      // Task 57-c 二次确认开关：开（默认）→ 建待入场等下一根收盘 K 线复核；
  1082→      // 关 → 同 tick 直接开仓（executeOpen 内容量/同币/资金/幂等全量守卫，快 1 根 K 线）
  1083→      if ((cfg.waitBarConfirm ?? 1) === 1) {
  1084→        // 创建待入场记录（每轮次一条，数据库主键兜底幂等）；等待下一根已收盘 K 线复核
  1085→        const claimed = await this.repo.claimPending({
  1086→          cycleId: rKey,
  1087→          symbol: signal.symbol,
  1088→          side,
  1089→          firstBarTime: signal.lastBarTime ?? Date.now(),
  1090→          firstSeenAt: Date.now(),
  1091→          snapshotJson: JSON.stringify({ signal, modeTag: verdict.modeTag, reason: verdict.reason }),
  1092→          status: 'ACTIVE',
  1093→          cancelReason: null,
  1094→          openPositionId: null,
  1095→          createdAt: Date.now(),
  1096→          updatedAt: Date.now(),
  1097→        });
  1098→        if (claimed) {
  1099→          pendingSymbols.add(signal.symbol); // 同轮循环内容量即时收紧
  1100→          logDecision(
  1101→            `待入场创建（strategy=${this.strategy.id} mode=${verdict.modeTag} round=${round}，等待下一根已收盘 K 线复核）` +
  1102→              `symbol=${signal.symbol} side=${side} posCycle=${cycleId} firstBar=${signal.lastBarTime} ` +
  1103→              `score=${signal.score.toFixed(0)} er=${signal.features.efficiencyRatio.toFixed(2)} ` +
  1104→              `htf5m=${signal.htf5m.dir} htf15m=${signal.htf15m.dir} verdict=${verdict.reason}`,
  1105→          );
  1106→        }
  1107→      } else {
  1108→        const r = await this.executeOpen({ signal, rKey, side, verdict, cfg, eq, via: 'direct' });
  1109→        if (r.outcome === 'opened') pendingSymbols.add(signal.symbol); // 同轮循环内容量即时收紧

  1152→  private async accountEquity(accId: string): Promise<{
  1153→    balance: number;
  1154→    marginUsed: number;
  1155→    unrealized: number;
  1156→    equity: number;
  1157→    openCount: number;
  1158→  }> {

  1160→    const open = await db.paperPosition.findMany({ where: { accountId: accId, status: 'OPEN' } });
  1161→    let marginUsed = 0;
  1162→    let unrealized = 0;
  1163→    for (const p of open) {
  1164→      const mark = this.deps.markPrice(p.symbol);
  1165→      const m = mark > 0 ? mark : p.entryPrice;
  1166→      marginUsed += p.margin;
  1167→      unrealized += (m - p.entryPrice) * p.qty * (p.side === 'LONG' ? 1 : -1);
  1168→    }
  1169→    const balance = acc?.balance ?? 0;
  1170→    return {
  1171→      balance,
  1172→      marginUsed,
  1173→      unrealized,
  1174→      equity: balance + marginUsed + unrealized,
  1175→      openCount: open.length,
  1176→    };
  1177→  }

  1235→  async closeOne(positionId: string): Promise<{ ok: boolean; message: string }> {
  1236→    const meta = await this.repo.getMeta(positionId);
  1237→    if (!meta || meta.status !== 'OPEN') return { ok: false, message: '持仓不存在或已平仓' };
  1238→    const signal = this.deps.getSignals().find((s) => s.symbol === meta.symbol) ?? null;
  1239→    await this.closePosition(meta, 'MANUAL', signal);
  1240→    return { ok: true, message: '已提交平仓' };

  1243→  async closeAllManual(): Promise<{ closed: number; skipped: number; totalPnlWithSlip: number }> {
  1244→    const openMetas = await this.repo.loadMetasByStatus('OPEN');
  1245→    const signals = this.deps.getSignals();
  1246→    let closed = 0;
  1247→    let skipped = 0;
  1248→    let total = 0;
  1249→    for (const meta of openMetas) {
  1250→      const signal = signals.find((s) => s.symbol === meta.symbol) ?? null;
  1251→      const before = await db.paperPosition.findUnique({ where: { id: meta.positionId } });
  1252→      if (!before || before.status !== 'OPEN') {
  1253→        skipped++;
  1254→        continue;
  1255→      }
  1256→      await this.closePosition(meta, 'MANUAL', signal);
  1257→      const after = await db.paperPosition.findUnique({ where: { id: meta.positionId } });
  1258→      if (after && after.status !== 'OPEN') {
  1259→        closed++;
  1260→        total += after.pnl ?? 0;
  1261→      } else {
  1262→        skipped++;
  1263→      }
  1264→    }
  1265→    return { closed, skipped, totalPnlWithSlip: total };
  1266→  }
