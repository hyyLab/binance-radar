   228→   228→      });

   231→   231→      s.importState(await this.repo.loadStrategyStates(s.id));
   232→   232→    } catch {
   233→   233→      // 恢复失败从空状态开始（轮次按首次处理，保守不丢单）
   234→   234→    }

   257→   257→  async setRuntimeParams(patch: {
   258→   258→    strategyParams?: Record<string, unknown>;

   266→   266→    const dbPatch: {
   267→   267→      strategyParamsJson?: string | null;
   268→   268→      waitBarConfirm?: number;
   269→   269→      cooldownMs?: number | null;
   270→   270→      marginRatio?: number;
   271→   271→    } = {};
   272→   272→    let strategyChanged = false;
   273→   273→    if (patch.strategyParams !== undefined) {
   274→   274→      const map = parseStrategyParamsMap(cfg?.strategyParamsJson);
   275→   275→      map[strategyId] = patch.strategyParams;
   276→   276→      dbPatch.strategyParamsJson = JSON.stringify(map);

   278→   278→    }

   284→   284→    if (strategyChanged && !this.fixedStrategy) {
   285→   285→      await this.applyStrategy(strategyId); // 热装配：新参数立即用于后续评估；轮次状态从 DB 恢复
   286→   286→      logDecision(`参数控制台：策略参数已更新并热应用（strategy=${strategyId} params=${JSON.stringify(patch.strategyParams)}）`);
   287→   287→    }
   288→   288→    // marginRatio 为基础设施列：每 tick 重读 cfg，下一 tick 生效（无需重装配）
   289→   289→    if (patch.marginRatio !== undefined) {
   290→   290→      logDecision(`参数控制台：每仓保证金比例已更新为 ${patch.marginRatio}（下一 tick 生效；实际比例 = min(设定值, 95%÷持仓数)）`);
   291→   291→    }
   292→   292→  }

   311→   311→      await this.applyStrategy(strategyId);
   312→   312→      logDecision(`参数控制台：已恢复全部参数为代码默认（strategy=${strategyId}）`);

   323→   323→  /**
   324→   324→   * 崩溃窗口修复（任务 §七：服务重启后必须恢复持仓关联信息）：
   325→   325→   * a) 自动账户 OPEN 持仓缺 TrendTradeMeta → 从最近一条已抢占未绑定的 OPEN 决策恢复
   326→   326→   *    cycleId；找不到则标 ORPHAN 周期（保守：只受强平/人工/反向确认管理——ORPHAN 周期
   327→   327→   *    无从与 ENDED 事件做周期匹配，按 §2.2 不再「仅凭 symbol」离场）。
   328→   328→   * b) meta OPEN 但 PaperPosition 已 CLOSED/LIQUIDATED → 补写平仓终态。
   329→   329→   */
   330→   330→  private async reconcile(): Promise<void> {
   331→   331→    try {
   332→   332→      const acc = await db.paperAccount.findUnique({ where: { name: TREND_TRADE.ACCOUNT_NAME } });
   333→   333→      if (!acc) return;
   334→   334→      const openPositions = await db.paperPosition.findMany({
   335→   335→        where: { accountId: acc.id, status: 'OPEN' },
   336→   336→      });
   337→   337→      for (const p of openPositions) {
   338→   338→        const meta = await this.repo.getMeta(p.id);
   339→   339→        if (!meta) {
   340→   340→          const claimed = await db.$queryRawUnsafe<{ cycleId: string; createdAt: Date }[]>(
   341→   341→            `SELECT cycleId, createdAt FROM TrendAutoDecision
   342→   342→             WHERE symbol = ? AND action = 'OPEN' AND positionId IS NULL
   343→   343→             ORDER BY createdAt DESC LIMIT 1`,
   344→   344→            p.symbol,
   345→   345→          );
   346→   346→          const cycleId = claimed[0]?.cycleId ?? `ORPHAN:${p.symbol}:${p.openedAt.getTime()}`;
   347→   347→          await this.repo.upsertMeta({
   348→   348→            positionId: p.id,
   349→   349→            trendCycleId: cycleId,
   350→   350→            symbol: p.symbol,
   351→   351→            side: p.side,
   352→   352→            trendPhaseAtOpen: 'CONFIRMED_RESTORED',
   353→   353→            scoreAtOpen: 0,
   354→   354→            featuresAtOpen: JSON.stringify({ restored: true, recoveredAt: Date.now() }),
   355→   355→            signalAtOpen: '{}',
   356→   356→            slippageEst: p.qty * p.entryPrice * (TREND_TRADE.SLIPPAGE_BPS / 10_000),
   357→   357→            status: 'OPEN',
   358→   358→            exitReason: null,
   359→   359→            exitSignalJson: null,
   360→   360→            netPnlWithSlip: null,
   361→   361→            openDecisionAt: claimed[0]?.createdAt ?? p.openedAt,
   362→   362→            closedAt: null,
   363→   363→          });
   364→   364→        }
   365→   365→      }
   366→   366→      // meta OPEN 但仓位已终态 → 补写
   367→   367→      const openMetas = await this.repo.loadMetasByStatus('OPEN');
   368→   368→      for (const m of openMetas) {
   369→   369→        const pos = await db.paperPosition.findUnique({ where: { id: m.positionId } });
   370→   370→        if (!pos) {
   371→   371→          await this.repo.upsertMeta({ ...m, status: 'CLOSED', exitReason: 'MANUAL', closedAt: new Date() });
   372→   372→          continue;
   373→   373→        }
   374→   374→        if (pos.status === 'LIQUIDATED') {
   375→   375→          await this.finalizeLiquidated(m, pos.margin, pos.pnl ?? -pos.margin);
   376→   376→        } else if (pos.status === 'CLOSED') {
   377→   377→          await this.repo.upsertMeta({
   378→   378→            ...m,
   379→   379→            status: 'CLOSED',
   380→   380→            exitReason: 'MANUAL',
   381→   381→            netPnlWithSlip: (pos.pnl ?? 0) - m.slippageEst,
   382→   382→            closedAt: pos.closedAt ?? new Date(),
   383→   383→          });
   384→   384→        }
   385→   385→      }
   386→   386→    } catch (e) {
   387→   387→      this.lastError = e instanceof Error ? e.message : String(e);
   388→   388→      console.warn('[trend-auto] reconcile 失败：', this.lastError);
   389→   389→    }
   390→   390→  }
   391→   391→
   392→   392→  // ============ 主循环 ============
   393→   393→
   394→   394→  /** 单步主循环：public 以便测试/回放驱动（生产由 SCAN_MS 定时器调用） */
   395→   395→  // 重入保护：setInterval 不等待异步完成——上一轮 tick 未跑完（行情/DB 慢）时下一轮并发
   396→   396→  // 执行，会按同一份旧余额重复开仓。60s 僵死兜底：异常挂起时不永久卡死主循环。
   397→   397→  private ticking = false;
   398→   398→  private tickStartedAt = 0;
   399→   399→
   400→   400→  async tick(): Promise<void> {
   401→   401→    if (this.stopped) return;
   402→   402→    if (this.ticking && Date.now() - this.tickStartedAt < 60_000) return;
   403→   403→    this.ticking = true;
   404→   404→    this.tickStartedAt = Date.now();
   405→   405→    try {
   406→   406→      const now = Date.now();
   407→   407→      this.lastTickAt = now;
   408→   408→      const cfg = await this.repo.loadConfig(TREND_TRADE.ACCOUNT_NAME);
   409→   409→      if (!cfg) return;
   410→   410→      const acc = await db.paperAccount.findUnique({ where: { name: TREND_TRADE.ACCOUNT_NAME } });
   411→   411→      if (!acc) return;
   412→   412→      const accId = acc.id;
   413→   413→
   414→   414→      // ① 持续风险巡检（任何模式都执行——强平不因暂停/数据不可用而缺席）
   415→   415→      await this.engine.liquidationCheck(accId);
   416→   416→      await this.finalizeNewLiquidations();
   417→   417→
   418→   418→      // ② 持仓结果采集（§七：MFE/MAE 按当前可用标记价增量更新——任何模式都执行）
   419→   419→      await this.updateOpenStats();
   420→   420→
   421→   421→      // ②b 策略自愈：DB 配置的算法与实例不一致（如 control 已写库但实例切换失败）→ 重新装配
   422→   422→      if (!this.fixedStrategy && cfg.strategyId && cfg.strategyId !== this.strategy.id) {
   423→   423→        await this.applyStrategy(cfg.strategyId);
   424→   424→        logDecision(`策略自愈切换 → ${cfg.strategyId}`);
   425→   425→      }
   426→   426→
   427→   427→      // ③ 退出管理（暂停模式也执行：趋势结束/反向确认/策略离场照常执行）
   428→   428→      const signals = this.deps.getSignals();
   429→   429→      const bySymbol = new Map(signals.map((s) => [s.symbol, s]));
   430→   430→      await this.manageExits(bySymbol);
   431→   431→
   432→   432→      // ③b 策略行情采样（冲高检测等需要短窗价格序列；只喂当时可用的标记价）
   433→   433→      for (const s of signals) {
   434→   434→        const mark = this.deps.markPrice(s.symbol);
   435→   435→        if (mark > 0) this.strategy.observe(s.symbol, mark, now);
   436→   436→      }
   437→   437→      for (const meta of await this.repo.loadMetasByStatus('OPEN')) {
   438→   438→        const mark = this.deps.markPrice(meta.symbol);
   439→   439→        if (mark > 0) this.strategy.observe(meta.symbol, mark, now);
   440→   440→      }
   441→   441→
   442→   442→      // ④ 权益峰值（回撤基准；任何模式都更新）
   443→   443→      const eq = await this.accountEquity(accId);
   444→   444→      if (eq.equity > cfg.peakEquity) {
   445→   445→        await this.repo.updatePeakEquity(TREND_TRADE.ACCOUNT_NAME, eq.equity);
   446→   446→      }
   447→   447→
   448→   448→      // ⑤ 开仓（仅 RUNNING 且数据新鲜；暂停/停止只停新开仓与待入场转换）
   449→   449→      if (cfg.mode === 'RUNNING') {
   450→   450→        await this.managePendings(bySymbol, cfg, eq); // 先转换已到期的待入场（等一根 K 线复核）
   451→   451→        await this.manageEntries(cfg, signals, eq); // 再为策略想开仓的趋势创建待入场记录
   452→   452→      }
   453→   453→      this.lastError = null;
   454→   454→    } catch (e) {
   455→   455→      this.lastError = e instanceof Error ? e.message : String(e);
   456→   456→      console.warn('[trend-auto] tick 失败：', this.lastError);
   457→   457→    } finally {
   458→   458→      this.ticking = false;
   459→   459→    }
   460→   460→  }
   461→   461→
   462→   462→  // ============ 持仓结果采集（Task 56 §七） ============
   463→   463→
   464→   464→  /**
   465→   465→   * MFE/MAE 增量更新：只用当前 tick 可得的标记价（多源供价的当前快照），逐 tick 维护
   466→   466→   * 「持仓期间」的最大有利/不利浮动盈亏（占初始保证金 %）——绝不使用事后才知道的
   467→   467→   * 最高/最低价构造当时不存在的数据。MFE 更新时刻与峰值后回撤一并落库。
   468→   468→   */
   469→   469→  private async updateOpenStats(): Promise<void> {
   470→   470→    const openMetas = await this.repo.loadMetasByStatus('OPEN');
   471→   471→    for (const meta of openMetas) {
   472→   472→      try {
   473→   473→        const pos = await db.paperPosition.findUnique({ where: { id: meta.positionId } });
   474→   474→        if (!pos || pos.status !== 'OPEN' || !(pos.margin > 0)) continue;
   475→   475→        const mark = this.deps.markPrice(meta.symbol);
   476→   476→        if (!(mark > 0)) continue; // 标记价不可用：跳过本 tick（不用过期价格虚构统计）
   477→   477→        const sign = meta.side === 'LONG' ? 1 : -1;
   478→   478→        const pnlPct = (((mark - pos.entryPrice) * pos.qty * sign) / pos.margin) * 100;
   479→   479→        const prevMfe = meta.mfe ?? null;
   480→   480→        const mfe = prevMfe != null ? Math.max(prevMfe, pnlPct) : pnlPct;
   481→   481→        const mae = meta.mae != null ? Math.min(meta.mae, pnlPct) : pnlPct;
   482→   482→        const isNewPeak = prevMfe == null || pnlPct > prevMfe;
   483→   483→        const mfeAt = isNewPeak ? Date.now() : (meta.mfeAt ?? null);
   484→   484→        const mfeDrawdown = mfe - pnlPct;
   485→   485→        if (
   486→   486→          prevMfe == null ||
   487→   487→          meta.mae == null ||
   488→   488→          Math.abs(mfe - prevMfe) > 1e-9 ||
   489→   489→          Math.abs(mae - meta.mae) > 1e-9
   490→   490→        ) {
   491→   491→          await this.repo.updateTradeStats(meta.positionId, { mfe, mae, mfeAt, mfeDrawdown });
   492→   492→        }
   493→   493→      } catch {
   494→   494→        // 单笔统计失败不影响主循环
   495→   495→      }
   496→   496→    }
   497→   497→  }
   498→   498→
   499→   499→  // ============ 退出管理 ============
   500→   500→
   501→   501→  /**
   502→   502→   * 退出管理 v4（Task 57：基础设施硬规则优先，策略离场在其内表达）：
   503→   503→   * ① ENDED（lifecycle 已按「连续 N 根不同已收盘 K 线持续证据」确认）→ 平仓，但必须：
   504→   504→   *    a) 信号携带的 trendCycleId 与持仓绑定周期一致（旧周期 ENDED 不杀新周期持仓，
   505→   505→   *       也不仅凭 symbol 平仓）；
   506→   506→   *    b) 信号最近 K 线晚于开仓依据的 K 线（开仓所在 K 线不得作为该持仓的结束证据）。
   507→   507→   * ② END_PENDING（同周期出现结束迹象）→ 只记录首次报警时刻（firstEndSignAt）；
   508→   508→   *    是否提前离场交给策略（短周期套利：走弱先走，走强后按轮次再入场）。
   509→   509→   * ③ 策略离场（evaluateExit）：SPIKE_TAKE 冲高套利 / WEAKNESS_EARLY 走弱提前离场——
   510→   510→   *    基于当时可用的标记价短窗与信号相位，不使用未来数据。
   511→   511→   * ④ 反向趋势已确认（CONFIRMED_*，方向与持仓相反，dataFresh）→ 立即平仓。
   512→   512→   * 数据缺失（DATA_UNAVAILABLE）永不触发离场；信号缺失（退池）不误平仓，仅强平/人工管理。
   513→   513→   */
   514→   514→  private async manageExits(bySymbol: Map<string, UnifiedTrendSignal>): Promise<void> {
   515→   515→    const openMetas = await this.repo.loadMetasByStatus('OPEN');
   516→   516→    for (const meta of openMetas) {
   517→   517→      const signal = bySymbol.get(meta.symbol);
   518→   518→      if (!signal) continue; // 信号缺失（标的退池/未评估）：不误平仓，仅强平/人工管理
   519→   519→      const favorable = meta.side === 'LONG' ? signal.direction === 'up' : signal.direction === 'down';
   520→   520→
   521→   521→      // ① 趋势已结束（ENDED）→ 周期匹配 + 证据时序校验后自行平仓
   522→   522→      if (signal.phase === 'ENDED') {
   523→   523→        if (!signal.trendCycleId || signal.trendCycleId !== meta.trendCycleId) {
   524→   524→          // 旧周期 ENDED 归属他人——不得据此平掉本周期持仓
   525→   525→          logDecision(
   526→   526→            `离场跳过 ENDED 周期不匹配 symbol=${meta.symbol} posCycle=${meta.trendCycleId} ` +
   527→   527→              `signalCycle=${signal.trendCycleId ?? '-'} positionId=${meta.positionId}`,
   528→   528→          );
   529→   529→          continue;
   530→   530→        }
   531→   531→        const entryBar = this.entryBarTimeOf(meta);
   532→   532→        if (entryBar != null && signal.lastBarTime != null && signal.lastBarTime <= entryBar) {
   533→   533→          // 结束证据未出现「开仓之后的新收盘 K 线」——不得平仓
   534→   534→          logDecision(
   535→   535→            `离场跳过 证据K线未晚于开仓K线 symbol=${meta.symbol} posCycle=${meta.trendCycleId} ` +
   536→   536→              `entryBar=${entryBar} signalBar=${signal.lastBarTime} positionId=${meta.positionId}`,
   537→   537→          );
   538→   538→          continue;
   539→   539→        }
   540→   540→        await this.closePosition(meta, 'TREND_ENDED', signal);
   541→   541→        continue;
   542→   542→      }
   543→   543→
   544→   544→      // ② 本周期出现结束迹象（模型已报警）→ 只记录首次报警时刻；是否提前离场由策略决定
   545→   545→      if (signal.phase === 'END_PENDING' && signal.trendCycleId === meta.trendCycleId) {
   546→   546→        if (signal.endPendingSince != null && meta.firstEndSignAt == null) {
   547→   547→          await this.repo.markFirstEndSign(meta.positionId, signal.endPendingSince);
   548→   548→          logDecision(
   549→   549→            `结束迹象首次报警（是否提前离场由策略「${this.strategy.name}」决定） ` +
   550→   550→              `symbol=${meta.symbol} posCycle=${meta.trendCycleId} ev=${signal.endEvidenceBars} ` +
   551→   551→              `positionId=${meta.positionId}`,
   552→   552→          );
   553→   553→        }
   554→   554→      }
   555→   555→
   556→   556→      // ③ 策略离场（SPIKE_TAKE 冲高套利 / WEAKNESS_EARLY 走弱先走；基于当时可用数据）
   557→   557→      const mark = this.deps.markPrice(meta.symbol);
   558→   558→      if (mark > 0) {
   559→   559→        const pos = await db.paperPosition.findUnique({ where: { id: meta.positionId } });
   560→   560→        if (pos && pos.status === 'OPEN') {
   561→   561→          const v = this.strategy.evaluateExit({
   562→   562→            meta,
   563→   563→            pos: {
   564→   564→              entryPrice: pos.entryPrice,
   565→   565→              qty: pos.qty,
   566→   566→              margin: pos.margin,
   567→   567→              side: meta.side as 'LONG' | 'SHORT',
   568→   568→              openedAt: pos.openedAt instanceof Date ? pos.openedAt.getTime() : Number(pos.openedAt),
   569→   569→            },
   570→   570→            mark,
   571→   571→            signal,
   572→   572→            now: Date.now(),
   573→   573→          });
   574→   574→          if (v.exit && v.reason) {
   575→   575→            logDecision(`策略离场 [${this.strategy.id}] ${v.detail} positionId=${meta.positionId}`);
   576→   576→            await this.closePosition(meta, v.reason, signal);
   577→   577→            continue;
   578→   578→          }
   579→   579→        }
   580→   580→      }
   581→   581→
   582→   582→      // ④ 反向趋势已确认（独立确认 + 数据新鲜）→ 自行平仓（反手开仓见 manageEntries 守卫）
   583→   583→      if (
   584→   584→        (signal.phase === 'CONFIRMED_UP' || signal.phase === 'CONFIRMED_DOWN') &&
   585→   585→        !favorable &&
   586→   586→        signal.dataFresh
   587→   587→      ) {
   588→   588→        await this.closePosition(meta, 'REVERSE_CONFIRMED', signal);
   589→   589→      }
   590→   590→    }
   591→   591→  }

   593→   593→  /** 开仓依据的已收盘 K 线（§2.3 守卫用；旧数据缺列时回退到 signalAtOpen.lastBarTime） */
   594→   594→  private entryBarTimeOf(meta: TrendTradeMetaRow): number | null {
   595→   595→    if (meta.entryBarTime != null) return meta.entryBarTime;
   596→   596→    try {
   597→   597→      const s = JSON.parse(meta.signalAtOpen) as { lastBarTime?: number | null };
   598→   598→      return typeof s.lastBarTime === 'number' ? s.lastBarTime : null;
   599→   599→    } catch {
   600→   600→      return null;
   601→   601→    }
   602→   602→  }

   604→   604→  /** 平仓（幂等护栏 = 仓位状态本身；决策行事后审计记录） */
   605→   605→  private async closePosition(
   606→   606→    metaIn: TrendTradeMetaRow,
   607→   607→    reason: TrendExitReason,
   608→   608→    signal: UnifiedTrendSignal | null,
   609→   609→  ): Promise<void> {
   610→   610→    try {
   611→   611→      // 重读最新 meta（同轮 tick 内 ② 可能已写入 firstEndSignAt——避免旧快照覆盖回 null）
   612→   612→      const meta = (await this.repo.getMeta(metaIn.positionId)) ?? metaIn;
   613→   613→      const pos = await db.paperPosition.findUnique({ where: { id: meta.positionId } });
   614→   614→      if (!pos || pos.status !== 'OPEN') {
   615→   615→        // 已终态（并发/重启窗口）：补写 meta 终态
   616→   616→        await this.repo.upsertMeta({
   617→   617→          ...meta,
   618→   618→          status: 'CLOSED',
   619→   619→          exitReason: pos?.status === 'LIQUIDATED' ? 'LIQUIDATED' : reason,
   620→   620→          netPnlWithSlip: (pos?.pnl ?? 0) - meta.slippageEst,
   621→   621→          closedAt: pos?.closedAt ?? new Date(),
   622→   622→        });
   623→   623→        return;
   624→   624→      }
   625→   625→      const r = await this.engine.close(meta.positionId);
   626→   626→      const mark = r.price;
   627→   627→      const closeSlip = mark * pos.qty * (TREND_TRADE.SLIPPAGE_BPS / 10_000);
   628→   628→      const netWithSlip = r.pnl - meta.slippageEst - closeSlip;
   629→   629→      await this.repo.upsertMeta({
   630→   630→        ...meta,
   631→   631→        status: 'CLOSED',
   632→   632→        exitReason: reason,
   633→   633→        exitSignalJson: signal ? JSON.stringify(signal) : null,
   634→   634→        netPnlWithSlip: netWithSlip,
   635→   635→        closedAt: new Date(),
   636→   636→        exitEndReason: signal?.endReason ?? meta.exitEndReason ?? null,
   637→   637→        exitEvidenceBars: signal?.endEvidenceBars ?? meta.exitEvidenceBars ?? null,
   638→   638→        closeNotional: mark * pos.qty,
   639→   639→      });
   640→   640→      const round = meta.entryRound ?? 1;
   641→   641→      await this.repo.claimDecision({
   642→   642→        id: `${roundKey(meta.trendCycleId, round)}:CLOSE`,
   643→   643→        symbol: meta.symbol,
   644→   644→        action: reason === 'REVERSE_CONFIRMED' ? 'REVERSE_CLOSE' : 'CLOSE',
   645→   645→        cycleId: meta.trendCycleId,
   646→   646→        positionId: meta.positionId,
   647→   647→        detail: JSON.stringify({
   648→   648→          reason,
   649→   649→          round,
   650→   650→          strategyId: this.strategy.id,
   651→   651→          exitEndReason: signal?.endReason ?? null,
   652→   652→          exitEvidenceBars: signal?.endEvidenceBars ?? null,
   653→   653→          exitPrice: r.price,
   654→   654→          netPnl: r.pnl,
   655→   655→          netPnlWithSlip: netWithSlip,
   656→   656→          mfe: meta.mfe ?? null,
   657→   657→          mae: meta.mae ?? null,
   658→   658→          endReason: signal?.endReason ?? null,
   659→   659→        }),
   660→   660→      });
   661→   661→      // 策略回调：保留轮次状态供「走强再入场」判定
   662→   662→      this.strategy.onClose(meta.symbol, meta.trendCycleId, reason, Date.now());
   663→   663→      this.closes++;
   664→   664→      logDecision(
   665→   665→        `平仓 reason=${reason} strategy=${this.strategy.id} round=${round} endReason=${signal?.endReason ?? '-'} symbol=${meta.symbol} ` +
   666→   666→          `side=${meta.side} posCycle=${meta.trendCycleId} positionId=${meta.positionId} ` +
   667→   667→          `ev=${signal?.endEvidenceBars ?? '-'} exitPrice=${mark} netPnl=${r.pnl.toFixed(4)} ` +
   668→   668→          `netWithSlip=${netWithSlip.toFixed(4)} mfe=${meta.mfe?.toFixed(1) ?? '-'}% mae=${meta.mae?.toFixed(1) ?? '-'}%`,
   669→   669→      );
   670→   670→    } catch (e) {
   671→   671→      // 平仓暂时性失败（行情不可用等）：下一轮重试（决策行未写，不烧掉本周期）
   672→   672→      // ⚠️ 此处只能用入参 metaIn——重读的 meta 声明在 try 作用域内，catch 引用会 ReferenceError
   673→   673→      if (!(e instanceof PaperError)) {
   674→   674→        this.pushSkip(metaIn.symbol, `平仓异常：${e instanceof Error ? e.message : e}`, metaIn.trendCycleId);
   675→   675→      }
   676→   676→    }
   677→   677→  }

   679→   679→  /** 强平巡检后的 meta 终态补写（强平是执行结果，如实记录、不伪装模型止损） */

   684→   684→      if (pos && pos.status === 'LIQUIDATED') {

   717→   717→  /** 策略入场评估上下文（快照；策略不得改动） */
   718→   718→  private strategyCtx(
   719→   719→    now: number,
   720→   720→    mark: number,
   721→   721→    cfg: TrendAutoConfigRow,
   722→   722→    eq: { balance: number; marginUsed: number; unrealized: number; equity: number; openCount: number },

   729→   729→  /** 开仓时高周期门控状态标签（快照存档用；方向分级与策略同源） */
   730→   730→  private htfGateLabel(signal: UnifiedTrendSignal): HtfGateState {
   731→   731→    const trendDir = signal.direction === 'down' ? 'down' : 'up';
   732→   732→    const g5 = classifyHtf(signal.htf5m.dir, signal.htf5m.valid, trendDir);
   733→   733→    const g15 = classifyHtf(signal.htf15m.dir, signal.htf15m.valid, trendDir);
   734→   734→    if (g5 === 'INVALID' || g15 === 'INVALID') return 'INVALID';
