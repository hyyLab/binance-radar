   117→   117→  // ============ 生命周期 ============

   119→   119→  async ensureStarted(): Promise<void> {
   120→   120→    if (this.started) return;
   121→   121→    if (this.starting) {
   122→   122→      await Promise.race([this.starting, new Promise((r) => setTimeout(r, 6000))]);
   123→   123→      return;
   124→   124→    }
   125→   125→    this.started = true;
   126→   126→    this.stopped = false;
   127→   127→    this.startedAt = Date.now();
   128→   128→    this.starting = this.boot().finally(() => {
   129→   129→      this.starting = null;
   130→   130→    });
   131→   131→    await Promise.race([this.starting, new Promise((r) => setTimeout(r, 6000))]);
   132→   132→  }

   134→   134→  private async boot(): Promise<void> {
   135→   135→    // 1) 建表（幂等，含 Task 56 扩展列/待入场表）+ 播种配置 + 独立账户初始化
   136→   136→    await ensureTrendTradeSchema();

   138→   138→    if (!seeded) {

   141→   141→        mode: 'RUNNING',
   142→   142→        initialCapital: TREND_TRADE.INITIAL_CAPITAL,

   149→   149→    }

   151→   151→    if (!acc) {
   152→   152→      await db.paperAccount.create({

   169→   169→  /**
   170→   170→   * 崩溃窗口修复（任务 §七：服务重启后必须恢复持仓关联信息）：
   171→   171→   * a) 自动账户 OPEN 持仓缺 TrendTradeMeta → 从最近一条已抢占未绑定的 OPEN 决策恢复
   172→   172→   *    cycleId；找不到则标 ORPHAN 周期（保守：只受强平/人工/反向确认管理——ORPHAN 周期
   173→   173→   *    无从与 ENDED 事件做周期匹配，按 §2.2 不再「仅凭 symbol」离场）。
   174→   174→   * b) meta OPEN 但 PaperPosition 已 CLOSED/LIQUIDATED → 补写平仓终态。
   175→   175→   */
   176→   176→  private async reconcile(): Promise<void> {
   177→   177→    try {
   178→   178→      const acc = await db.paperAccount.findUnique({ where: { name: TREND_TRADE.ACCOUNT_NAME } });
   179→   179→      if (!acc) return;
   180→   180→      const openPositions = await db.paperPosition.findMany({
   181→   181→        where: { accountId: acc.id, status: 'OPEN' },
   182→   182→      });
   183→   183→      for (const p of openPositions) {
   184→   184→        const meta = await this.repo.getMeta(p.id);
   185→   185→        if (!meta) {
   186→   186→          const claimed = await db.$queryRawUnsafe<{ cycleId: string; createdAt: Date }[]>(
   187→   187→            `SELECT cycleId, createdAt FROM TrendAutoDecision
   188→   188→             WHERE symbol = ? AND action = 'OPEN' AND positionId IS NULL
   189→   189→             ORDER BY createdAt DESC LIMIT 1`,
   190→   190→            p.symbol,
   191→   191→          );
   192→   192→          const cycleId = claimed[0]?.cycleId ?? `ORPHAN:${p.symbol}:${p.openedAt.getTime()}`;
   193→   193→          await this.repo.upsertMeta({
   194→   194→            positionId: p.id,
   195→   195→            trendCycleId: cycleId,
   196→   196→            symbol: p.symbol,
   197→   197→            side: p.side,
   198→   198→            trendPhaseAtOpen: 'CONFIRMED_RESTORED',
   199→   199→            scoreAtOpen: 0,
   200→   200→            featuresAtOpen: JSON.stringify({ restored: true, recoveredAt: Date.now() }),
   201→   201→            signalAtOpen: '{}',
   202→   202→            slippageEst: p.qty * p.entryPrice * (TREND_TRADE.SLIPPAGE_BPS / 10_000),
   203→   203→            status: 'OPEN',
   204→   204→            exitReason: null,
   205→   205→            exitSignalJson: null,
   206→   206→            netPnlWithSlip: null,
   207→   207→            openDecisionAt: claimed[0]?.createdAt ?? p.openedAt,
   208→   208→            closedAt: null,
   209→   209→          });
   210→   210→        }
   211→   211→      }
   212→   212→      // meta OPEN 但仓位已终态 → 补写
   213→   213→      const openMetas = await this.repo.loadMetasByStatus('OPEN');
   214→   214→      for (const m of openMetas) {
   215→   215→        const pos = await db.paperPosition.findUnique({ where: { id: m.positionId } });
   216→   216→        if (!pos) {
   217→   217→          await this.repo.upsertMeta({ ...m, status: 'CLOSED', exitReason: 'MANUAL', closedAt: new Date() });
   218→   218→          continue;
   219→   219→        }
   220→   220→        if (pos.status === 'LIQUIDATED') {
   221→   221→          await this.finalizeLiquidated(m, pos.margin, pos.pnl ?? -pos.margin);
   222→   222→        } else if (pos.status === 'CLOSED') {
   223→   223→          await this.repo.upsertMeta({
   224→   224→            ...m,
   225→   225→            status: 'CLOSED',
   226→   226→            exitReason: 'MANUAL',
   227→   227→            netPnlWithSlip: (pos.pnl ?? 0) - m.slippageEst,
   228→   228→            closedAt: pos.closedAt ?? new Date(),
   229→   229→          });
   230→   230→        }
   231→   231→      }
   232→   232→    } catch (e) {
   233→   233→      this.lastError = e instanceof Error ? e.message : String(e);
   234→   234→      console.warn('[trend-auto] reconcile 失败：', this.lastError);
   235→   235→    }
   236→   236→  }

   240→   240→  /** 单步主循环：public 以便测试/回放驱动（生产由 SCAN_MS 定时器调用） */
   241→   241→  // 重入保护：setInterval 不等待异步完成——上一轮 tick 未跑完（行情/DB 慢）时下一轮并发
   242→   242→  // 执行，会按同一份旧余额重复开仓。60s 僵死兜底：异常挂起时不永久卡死主循环。
   243→   243→  private ticking = false;
   244→   244→  private tickStartedAt = 0;
   245→   245→
   246→   246→  async tick(): Promise<void> {
   247→   247→    if (this.stopped) return;
   248→   248→    if (this.ticking && Date.now() - this.tickStartedAt < 60_000) return;
   249→   249→    this.ticking = true;
   250→   250→    this.tickStartedAt = Date.now();
   251→   251→    try {
   252→   252→      const now = Date.now();
   253→   253→      this.lastTickAt = now;
   254→   254→      const cfg = await this.repo.loadConfig(TREND_TRADE.ACCOUNT_NAME);
   255→   255→      if (!cfg) return;
   256→   256→      const acc = await db.paperAccount.findUnique({ where: { name: TREND_TRADE.ACCOUNT_NAME } });
   257→   257→      if (!acc) return;
   258→   258→      const accId = acc.id;
   259→   259→
   260→   260→      // ① 持续风险巡检（任何模式都执行——强平不因暂停/数据不可用而缺席）
   261→   261→      await this.engine.liquidationCheck(accId);
   262→   262→      await this.finalizeNewLiquidations();
   263→   263→
   264→   264→      // ② 持仓结果采集（§七：MFE/MAE 按当前可用标记价增量更新——任何模式都执行）
   265→   265→      await this.updateOpenStats();
   266→   266→
   267→   267→      // ③ 退出管理（暂停模式也执行：趋势结束/反向确认照常离场）
   268→   268→      const signals = this.deps.getSignals();
   269→   269→      const bySymbol = new Map(signals.map((s) => [s.symbol, s]));
   270→   270→      await this.manageExits(bySymbol);
   271→   271→
   272→   272→      // ④ 权益峰值（回撤基准；任何模式都更新）
   273→   273→      const eq = await this.accountEquity(accId);
   274→   274→      if (eq.equity > cfg.peakEquity) {
   275→   275→        await this.repo.updatePeakEquity(TREND_TRADE.ACCOUNT_NAME, eq.equity);
   276→   276→      }
   277→   277→
   278→   278→      // ⑤ 开仓（仅 RUNNING 且数据新鲜；暂停/停止只停新开仓与待入场转换）
   279→   279→      if (cfg.mode === 'RUNNING') {
   280→   280→        await this.managePendings(bySymbol, cfg, eq); // 先转换已到期的待入场（等一根 K 线复核）
   281→   281→        await this.manageEntries(cfg, signals, eq); // 再为新的确认趋势创建待入场记录
   282→   282→      }
   283→   283→      this.lastError = null;
   284→   284→    } catch (e) {
   285→   285→      this.lastError = e instanceof Error ? e.message : String(e);
   286→   286→      console.warn('[trend-auto] tick 失败：', this.lastError);
   287→   287→    } finally {
   288→   288→      this.ticking = false;
   289→   289→    }
   290→   290→  }
   291→   291→
   292→   292→  // ============ 持仓结果采集（Task 56 §七） ============
   293→   293→
   294→   294→  /**
   295→   295→   * MFE/MAE 增量更新：只用当前 tick 可得的标记价（多源供价的当前快照），逐 tick 维护
   296→   296→   * 「持仓期间」的最大有利/不利浮动盈亏（占初始保证金 %）——绝不使用事后才知道的
   297→   297→   * 最高/最低价构造当时不存在的数据。MFE 更新时刻与峰值后回撤一并落库。
   298→   298→   */
   299→   299→  private async updateOpenStats(): Promise<void> {
   300→   300→    const openMetas = await this.repo.loadMetasByStatus('OPEN');
   301→   301→    for (const meta of openMetas) {
   302→   302→      try {
   303→   303→        const pos = await db.paperPosition.findUnique({ where: { id: meta.positionId } });
   304→   304→        if (!pos || pos.status !== 'OPEN' || !(pos.margin > 0)) continue;
   305→   305→        const mark = this.deps.markPrice(meta.symbol);
   306→   306→        if (!(mark > 0)) continue; // 标记价不可用：跳过本 tick（不用过期价格虚构统计）
   307→   307→        const sign = meta.side === 'LONG' ? 1 : -1;
   308→   308→        const pnlPct = (((mark - pos.entryPrice) * pos.qty * sign) / pos.margin) * 100;
   309→   309→        const prevMfe = meta.mfe ?? null;
   310→   310→        const mfe = prevMfe != null ? Math.max(prevMfe, pnlPct) : pnlPct;
   311→   311→        const mae = meta.mae != null ? Math.min(meta.mae, pnlPct) : pnlPct;
   312→   312→        const isNewPeak = prevMfe == null || pnlPct > prevMfe;
   313→   313→        const mfeAt = isNewPeak ? Date.now() : (meta.mfeAt ?? null);
   314→   314→        const mfeDrawdown = mfe - pnlPct;
   315→   315→        if (
   316→   316→          prevMfe == null ||
   317→   317→          meta.mae == null ||
   318→   318→          Math.abs(mfe - prevMfe) > 1e-9 ||
   319→   319→          Math.abs(mae - meta.mae) > 1e-9
   320→   320→        ) {
   321→   321→          await this.repo.updateTradeStats(meta.positionId, { mfe, mae, mfeAt, mfeDrawdown });
   322→   322→        }
   323→   323→      } catch {
   324→   324→        // 单笔统计失败不影响主循环
   325→   325→      }
   326→   326→    }
   327→   327→  }

   330→   330→
   331→   331→  /**
   332→   332→   * 退出管理 v3（Task 56 用户定版：三根独立收盘 K 线结束确认 + 周期匹配 + 证据时序）：
   333→   333→   * ① ENDED（lifecycle 已按「连续 N 根不同已收盘 K 线持续证据」确认）→ 平仓，但必须：
   334→   334→   *    a) 信号携带的 trendCycleId 与持仓绑定周期一致（§2.2——旧周期 ENDED 不杀新周期持仓，
   335→   335→   *       也不仅凭 symbol 平仓）；
   336→   336→   *    b) 信号最近 K 线晚于开仓依据的 K 线（§2.3——开仓所在 K 线不得作为该持仓的结束证据）。
   337→   337→   * ② END_PENDING（同周期出现结束迹象）→ 只记录首次报警时刻（firstEndSignAt），不平仓；
   338→   338→   *    等 lifecycle 按三根 K 线确认。条件恢复时 lifecycle 自动清零重计（迟滞）。
   339→   339→   * ③ 反向趋势已确认（CONFIRMED_*，方向与持仓相反，dataFresh）→ 立即平仓
   340→   340→   *    （§3.2：独立确认的反向趋势可立即离场；反向候选/单根反向 K 线不够格）。
   341→   341→   * 数据缺失（DATA_UNAVAILABLE）永不触发离场；信号缺失（退池）不误平仓，仅强平/人工管理。
   342→   342→   */
   343→   343→  private async manageExits(bySymbol: Map<string, UnifiedTrendSignal>): Promise<void> {
   344→   344→    const openMetas = await this.repo.loadMetasByStatus('OPEN');
   345→   345→    for (const meta of openMetas) {
   346→   346→      const signal = bySymbol.get(meta.symbol);
   347→   347→      if (!signal) continue; // 信号缺失（标的退池/未评估）：不误平仓，仅强平/人工管理
   348→   348→      const favorable = meta.side === 'LONG' ? signal.direction === 'up' : signal.direction === 'down';
   349→   349→
   350→   350→      // ① 趋势已结束（ENDED）→ 周期匹配 + 证据时序校验后自行平仓
   351→   351→      if (signal.phase === 'ENDED') {
   352→   352→        if (!signal.trendCycleId || signal.trendCycleId !== meta.trendCycleId) {
   353→   353→          // §2.2：ENDED 归属于其他（旧）周期——不得据此平掉本周期持仓
   354→   354→          logDecision(
   355→   355→            `离场跳过 ENDED 周期不匹配 symbol=${meta.symbol} posCycle=${meta.trendCycleId} ` +
   356→   356→              `signalCycle=${signal.trendCycleId ?? '-'} positionId=${meta.positionId}`,
   357→   357→          );
   358→   358→          continue;
   359→   359→        }
   360→   360→        const entryBar = this.entryBarTimeOf(meta);
   361→   361→        if (entryBar != null && signal.lastBarTime != null && signal.lastBarTime <= entryBar) {
   362→   362→          // §2.3：结束证据未出现「开仓之后的新收盘 K 线」——不得平仓
   363→   363→          logDecision(
   364→   364→            `离场跳过 证据K线未晚于开仓K线 symbol=${meta.symbol} posCycle=${meta.trendCycleId} ` +
   365→   365→              `entryBar=${entryBar} signalBar=${signal.lastBarTime} positionId=${meta.positionId}`,
   366→   366→          );
   367→   367→          continue;
   368→   368→        }
   369→   369→        await this.closePosition(meta, 'TREND_ENDED', signal);
   370→   370→        continue;
   371→   371→      }
   372→   372→
   373→   373→      // ② 本周期出现结束迹象（模型已报警）→ 只记录首次报警时刻，等 N 根 K 线确认
   374→   374→      if (signal.phase === 'END_PENDING' && signal.trendCycleId === meta.trendCycleId) {
   375→   375→        if (signal.endPendingSince != null && meta.firstEndSignAt == null) {
   376→   376→          await this.repo.markFirstEndSign(meta.positionId, signal.endPendingSince);
   377→   377→          logDecision(
   378→   378→            `结束迹象首次报警（不平仓，等待 ${TREND_WATCH.END_CONFIRM_BARS} 根收盘 K 线确认） ` +
   379→   379→              `symbol=${meta.symbol} posCycle=${meta.trendCycleId} ev=${signal.endEvidenceBars} ` +
   380→   380→              `positionId=${meta.positionId}`,
   381→   381→          );
   382→   382→        }
   383→   383→        continue;
   384→   384→      }
   385→   385→
   386→   386→      // ③ 反向趋势已确认（独立确认 + 数据新鲜）→ 自行平仓（反手开仓见 manageEntries 守卫）
   387→   387→      if (
   388→   388→        (signal.phase === 'CONFIRMED_UP' || signal.phase === 'CONFIRMED_DOWN') &&
   389→   389→        !favorable &&
   390→   390→        signal.dataFresh
   391→   391→      ) {
   392→   392→        await this.closePosition(meta, 'REVERSE_CONFIRMED', signal);
   393→   393→      }
   394→   394→    }
   395→   395→  }
   396→   396→
   397→   397→  /** 开仓依据的已收盘 K 线（§2.3 守卫用；旧数据缺列时回退到 signalAtOpen.lastBarTime） */
   398→   398→  private entryBarTimeOf(meta: TrendTradeMetaRow): number | null {
   399→   399→    if (meta.entryBarTime != null) return meta.entryBarTime;
   400→   400→    try {
   401→   401→      const s = JSON.parse(meta.signalAtOpen) as { lastBarTime?: number | null };
   402→   402→      return typeof s.lastBarTime === 'number' ? s.lastBarTime : null;
   403→   403→    } catch {
   404→   404→      return null;
   405→   405→    }
   406→   406→  }

   408→   408→  /** 平仓（幂等护栏 = 仓位状态本身；决策行事后审计记录） */
   409→   409→  private async closePosition(
   410→   410→    meta: TrendTradeMetaRow,
   411→   411→    reason: TrendExitReason,
   412→   412→    signal: UnifiedTrendSignal | null,
   413→   413→  ): Promise<void> {
   414→   414→    try {
   415→   415→      const pos = await db.paperPosition.findUnique({ where: { id: meta.positionId } });
   416→   416→      if (!pos || pos.status !== 'OPEN') {
   417→   417→        // 已终态（并发/重启窗口）：补写 meta 终态
   418→   418→        await this.repo.upsertMeta({
   419→   419→          ...meta,
   420→   420→          status: 'CLOSED',
   421→   421→          exitReason: pos?.status === 'LIQUIDATED' ? 'LIQUIDATED' : reason,
   422→   422→          netPnlWithSlip: (pos?.pnl ?? 0) - meta.slippageEst,
   423→   423→          closedAt: pos?.closedAt ?? new Date(),
   424→   424→        });
   425→   425→        return;
   426→   426→      }
   427→   427→      const r = await this.engine.close(meta.positionId);
   428→   428→      const mark = r.price;
   429→   429→      const closeSlip = mark * pos.qty * (TREND_TRADE.SLIPPAGE_BPS / 10_000);
   430→   430→      const netWithSlip = r.pnl - meta.slippageEst - closeSlip;
   431→   431→      await this.repo.upsertMeta({
   432→   432→        ...meta,
   433→   433→        status: 'CLOSED',
   434→   434→        exitReason: reason,
   435→   435→        exitSignalJson: signal ? JSON.stringify(signal) : null,
   436→   436→        netPnlWithSlip: netWithSlip,
   437→   437→        closedAt: new Date(),
   438→   438→        exitEndReason: signal?.endReason ?? meta.exitEndReason ?? null,
   439→   439→        exitEvidenceBars: signal?.endEvidenceBars ?? meta.exitEvidenceBars ?? null,
   440→   440→        closeNotional: mark * pos.qty,
   441→   441→      });
   442→   442→      await this.repo.claimDecision({
   443→   443→        id: `${meta.trendCycleId}:CLOSE`,
   444→   444→        symbol: meta.symbol,
   445→   445→        action: reason === 'REVERSE_CONFIRMED' ? 'REVERSE_CLOSE' : 'CLOSE',
   446→   446→        cycleId: meta.trendCycleId,
   447→   447→        positionId: meta.positionId,
   448→   448→        detail: JSON.stringify({
   449→   449→          reason,
   450→   450→          exitEndReason: signal?.endReason ?? null,
   451→   451→          exitEvidenceBars: signal?.endEvidenceBars ?? null,
   452→   452→          exitPrice: r.price,
   453→   453→          netPnl: r.pnl,
   454→   454→          netPnlWithSlip: netWithSlip,
   455→   455→          mfe: meta.mfe ?? null,
   456→   456→          mae: meta.mae ?? null,
   457→   457→          endReason: signal?.endReason ?? null,
   458→   458→        }),
   459→   459→      });
   460→   460→      this.closes++;
   461→   461→      logDecision(
   462→   462→        `平仓 reason=${reason} endReason=${signal?.endReason ?? '-'} symbol=${meta.symbol} ` +
   463→   463→          `side=${meta.side} posCycle=${meta.trendCycleId} positionId=${meta.positionId} ` +
   464→   464→          `ev=${signal?.endEvidenceBars ?? '-'} exitPrice=${mark} netPnl=${r.pnl.toFixed(4)} ` +
   465→   465→          `netWithSlip=${netWithSlip.toFixed(4)} mfe=${meta.mfe?.toFixed(1) ?? '-'}% mae=${meta.mae?.toFixed(1) ?? '-'}%`,
   466→   466→      );
   467→   467→    } catch (e) {
   468→   468→      // 平仓暂时性失败（行情不可用等）：下一轮重试（决策行未写，不烧掉本周期）
   469→   469→      if (!(e instanceof PaperError)) {
   470→   470→        this.pushSkip(meta.symbol, `平仓异常：${e instanceof Error ? e.message : e}`, meta.trendCycleId);
   471→   471→      }
   472→   472→    }
   473→   473→  }

   509→   509→  // ============ 入场质量门槛（Task 56 §5.1/§5.2） ============
   510→   510→
   511→   511→  /**
   512→   512→   * 入场门槛（创建待入场与转换开仓共用同一判定；创建时初检、转换时全量复核）：
   513→   513→   * - score ≥ ENTRY_MIN_SCORE 且 1m ER ≥ ENTRY_MIN_ER（§5.1）
   514→   514→   * - 高周期过滤（§5.2）：5m 反对 → 拒绝；5m 中性或 15m 反对 → 严格门槛
   515→   515→   *   （score ≥ 85 且 ER ≥ 0.5）；高周期数据不足（INVALID）→ 拒绝（不得当作同向支持）
   516→   516→   * 返回 null = 通过；字符串 = 拒绝原因（审计/跳过记录用）。
   517→   517→   */
   518→   518→  private entryGateCheck(signal: UnifiedTrendSignal): string | null {
   519→   519→    const trendDir = signal.direction;
   520→   520→    if (trendDir !== 'up' && trendDir !== 'down') return '趋势方向缺失';
   521→   521→    const score = signal.score;
   522→   522→    const er = signal.features.efficiencyRatio;
   523→   523→    if (score < TREND_TRADE.ENTRY_MIN_SCORE) {
   524→   524→      return `评分 ${score.toFixed(0)} < ${TREND_TRADE.ENTRY_MIN_SCORE}`;
   525→   525→    }
   526→   526→    if (er < TREND_TRADE.ENTRY_MIN_ER) {
   527→   527→      return `ER ${er.toFixed(2)} < ${TREND_TRADE.ENTRY_MIN_ER}`;
   528→   528→    }
   529→   529→    const g5 = classifyHtf(signal.htf5m.dir, signal.htf5m.valid, trendDir);
   530→   530→    const g15 = classifyHtf(signal.htf15m.dir, signal.htf15m.valid, trendDir);
   531→   531→    if (g5 === 'INVALID' || g15 === 'INVALID') {
   532→   532→      return `高周期数据不足（5m=${g5}/15m=${g15}），不作同向支持`;
   533→   533→    }
   534→   534→    const strictOk = score >= TREND_TRADE.ENTRY_HTF_STRICT_SCORE && er >= TREND_TRADE.ENTRY_HTF_STRICT_ER;
   535→   535→    if (g5 === 'OPPOSED') {
   536→   536→      return '5m 方向反对 1m 趋势';
   537→   537→    }
   538→   538→    if (g5 === 'NEUTRAL') {
   539→   539→      if (g15 === 'OPPOSED') return '15m 方向反对且 5m 中性';
   540→   540→      if (!strictOk) {
   541→   541→        return `5m 中性未达严格门槛（score ${score.toFixed(0)}/${TREND_TRADE.ENTRY_HTF_STRICT_SCORE}，ER ${er.toFixed(2)}/${TREND_TRADE.ENTRY_HTF_STRICT_ER}）`;
   542→   542→      }
   543→   543→      return null;
   544→   544→    }
   545→   545→    // g5 === 'ALIGNED'
   546→   546→    if (g15 === 'OPPOSED' && !strictOk) {
   547→   547→      return `15m 反对且未达严格门槛（score ${score.toFixed(0)}/${TREND_TRADE.ENTRY_HTF_STRICT_SCORE}，ER ${er.toFixed(2)}/${TREND_TRADE.ENTRY_HTF_STRICT_ER}）`;
   548→   548→    }
   549→   549→    return null;
   550→   550→  }
   551→   551→
   552→   552→  /** 开仓时高周期门控状态标签（快照存档用；与 entryGateCheck 同口径） */
   553→   553→  private htfGateLabel(signal: UnifiedTrendSignal): HtfGateState {
   554→   554→    const trendDir = signal.direction === 'down' ? 'down' : 'up';
   555→   555→    const g5 = classifyHtf(signal.htf5m.dir, signal.htf5m.valid, trendDir);
   556→   556→    const g15 = classifyHtf(signal.htf15m.dir, signal.htf15m.valid, trendDir);
   557→   557→    if (g5 === 'INVALID' || g15 === 'INVALID') return 'INVALID';
   558→   558→    if (g5 === 'OPPOSED') return 'OPPOSED';
   559→   559→    if (g5 === 'NEUTRAL') return g15 === 'OPPOSED' ? 'OPPOSED' : 'NEUTRAL';
   560→   560→    return g15; // 5m ALIGNED：标签反映 15m 状态（ALIGNED/NEUTRAL/OPPOSED）
   561→   561→  }

   563→   563→  // ============ 开仓决策 ============
   564→   564→
   565→   565→  /**
   566→   566→   * 待入场管理（§5.3 二次确认）：每个 ACTIVE 待入场记录在出现「更新的已收盘 K 线」后
   567→   567→   * 全量复核（相位/周期/数据/质量门槛/冷却/容量/资金），通过才开仓；趋势失效、数据过期、
   568→   568→   * 门槛不再满足或记录过期 → 取消并记录原因。每周期至多一条记录、只允许成功开仓一次
   569→   569→   * （决策键 `{cycleId}:OPEN` 数据库唯一约束兜底）。
   570→   570→   */
   571→   571→  private async managePendings(
   572→   572→    bySymbol: Map<string, UnifiedTrendSignal>,
   573→   573→    cfg: TrendAutoConfigRow,
   574→   574→    eq: { balance: number; marginUsed: number; unrealized: number; equity: number; openCount: number },
   575→   575→  ): Promise<void> {
   576→   576→    const pendings = await this.repo.loadActivePendings();
   577→   577→    for (const p of pendings) {
   578→   578→      const signal = bySymbol.get(p.symbol);
   579→   579→      const cancel = async (reason: string): Promise<void> => {
   580→   580→        await this.repo.markPendingCanceled(p.cycleId, reason);
   581→   581→        logDecision(
   582→   582→          `待入场取消 cycle=${p.cycleId} symbol=${p.symbol} reason=${reason} ` +
   583→   583→            `ageMs=${Date.now() - p.firstSeenAt}`,
   584→   584→        );
   585→   585→      };
   586→   586→      // ① 记录过期（等待超时——趋势确认后长期未出现合格的新 K 线）
   587→   587→      if (Date.now() - p.firstSeenAt > TREND_TRADE.PENDING_EXPIRE_MS) {
   588→   588→        await cancel('待入场超时（等待新收盘 K 线复核超限）');
   589→   589→        continue;
   590→   590→      }
   591→   591→      // ② 信号缺失/数据过期/相位失效/周期变化 → 取消（§5.3：趋势失效或数据过期取消记录）
   592→   592→      if (!signal) {
   593→   593→        await cancel('信号缺失（标的退池/未评估）');
   594→   594→        continue;
   595→   595→      }
   596→   596→      if (!signal.dataFresh) {
   597→   597→        await cancel(`数据过期（${signal.dataUnavailableReason ?? '未知'}）`);
   598→   598→        continue;
   599→   599→      }
   600→   600→      if (
   601→   601→        !(TREND_TRADE.CONFIRMED_PHASES as readonly string[]).includes(signal.phase) ||
   602→   602→        signal.trendCycleId !== p.cycleId
   603→   603→      ) {
   604→   604→        await cancel(`趋势失效（phase=${signal.phase}，cycle=${signal.trendCycleId ?? '-'}）`);
   605→   605→        continue;
   606→   606→      }
   607→   607→      // ③ 还没等到「新的已收盘 K 线」→ 继续等待（§5.3 核心等待条件）
   608→   608→      if (signal.lastBarTime == null || signal.lastBarTime <= p.firstBarTime) continue;
   609→   609→      // ④ 冷却复核（§5.4：冷却以 symbol 为单位，不因周期变化绕过）
   610→   610→      const lastClose = await this.repo.lastCloseAt(p.symbol);
   611→   611→      if (lastClose != null && Date.now() - lastClose < TREND_TRADE.COOLDOWN_MS) {
   612→   612→        await cancel('转换时进入同币种冷却期');
   613→   613→        continue;
   614→   614→      }
   615→   615→      // ⑤ 质量门槛全量复核（趋势方向/阶段/评分/ER/高周期过滤）
   616→   616→      const gateReason = this.entryGateCheck(signal);
   617→   617→      if (gateReason) {
   618→   618→        await cancel(`复核未过门槛：${gateReason}`);
   619→   619→        continue;
   620→   620→      }
   621→   621→      // ⑥ 容量守卫（不满足时保留记录，等下轮复核直到过期）
   622→   622→      if (eq.openCount >= cfg.maxPositions) {
   623→   623→        this.pushSkip(p.symbol, `待入场转换等待容量（已达最大持仓数 ${cfg.maxPositions}）`, p.cycleId);
   624→   624→        continue;
   625→   625→      }
   626→   626→      // ⑦ 同 symbol 已有持仓（防御；正常由周期决策键保证）
   627→   627→      const dup = await db.paperPosition.findFirst({
   628→   628→        where: { accountId: await this.autoAccountId(), symbol: p.symbol, status: 'OPEN' },
   629→   629→      });
   630→   630→      if (dup) {
   631→   631→        await cancel('同币种已有持仓');
   632→   632→        continue;
   633→   633→      }
   634→   634→      // ⑧ 仓位测算（§六：每仓保证金 = 开仓时账户权益 × 20%；超出可用余额直接跳过）
   635→   635→      const feeFactor = 1 + cfg.leverage * TAKER_FEE;
   636→   636→      const targetMargin = floor2(eq.equity * cfg.marginRatio);
   637→   637→      const fee = targetMargin * cfg.leverage * TAKER_FEE;
   638→   638→      if (targetMargin + fee > eq.balance) {
   639→   639→        await this.recordSkipDb(
   640→   640→          p.symbol,
   641→   641→          p.cycleId,
   642→   642→          `权益 20% 保证金 ${targetMargin.toFixed(2)} + 手续费 ${fee.toFixed(2)} 超出可用余额 ${eq.balance.toFixed(2)}（跳过不变通）`,
   643→   643→        );
   644→   644→        await cancel('资金不足（权益 20% 保证金 + 手续费超出可用余额）');
   645→   645→        continue;
   646→   646→      }
   647→   647→      if (targetMargin < TREND_TRADE.MIN_MARGIN) {
   648→   648→        await this.recordSkipDb(
   649→   649→          p.symbol,
   650→   650→          p.cycleId,
   651→   651→          `权益 20% 保证金 ${targetMargin} 低于最小保证金 ${TREND_TRADE.MIN_MARGIN}`,
   652→   652→        );
   653→   653→        await cancel('目标保证金低于最小值');
   654→   654→        continue;
   655→   655→      }
   656→   656→      // ⑨ 幂等守卫（数据库唯一约束兜底）：抢占 `{cycleId}:OPEN` 决策

   667→   667→      }

   688→   688→            strength: signal.strength,
   689→   689→            anomaly: signal.anomaly,
   690→   690→            conflict: signal.conflict,
   691→   691→          }),
   692→   692→          signalAtOpen: JSON.stringify(signal),
   693→   693→          slippageEst,
   694→   694→          status: 'OPEN',
   695→   695→          exitReason: null,
   696→   696→          exitSignalJson: null,
   697→   697→          netPnlWithSlip: null,
   698→   698→          openDecisionAt: new Date(),
   699→   699→          closedAt: null,
   700→   700→          // Task 56 §七：开仓快照（可查询列）
   701→   701→          entryEr: signal.features.efficiencyRatio,
   702→   702→          htf5mDir: signal.htf5m.dir,

   746→   746→   * 新信号扫描（§5.3）：首次满足「确认相位 + 数据新鲜 + 质量门槛 + 冷却结束 + 容量 +
   747→   747→   * 独立性」的确认趋势，只创建待入场记录（等待下一根已收盘 K 线），不立即下单。
   748→   748→   */
   749→   749→  private async manageEntries(
   750→   750→    cfg: TrendAutoConfigRow,
   751→   751→    signals: UnifiedTrendSignal[],
   752→   752→    eq: { balance: number; marginUsed: number; unrealized: number; equity: number; openCount: number },
   753→   753→  ): Promise<void> {
   754→   754→    // 容量核算：已开仓 + 待入场（其 symbol 无持仓者，复核后将各占一仓）共同计入上限——
   755→   755→    // 防止待入场批量创建超过 maxPositions（转换层另有二次守卫）
   756→   756→    const accountId = await this.autoAccountId();
   757→   757→    const openRows = await db.paperPosition.findMany({ where: { accountId, status: 'OPEN' } });
   758→   758→    const openSymbols = new Set(openRows.map((p) => p.symbol));
   759→   759→    const activePendings = await this.repo.loadActivePendings();
   760→   760→    const pendingSymbols = new Set(
   761→   761→      activePendings.filter((p) => !openSymbols.has(p.symbol)).map((p) => p.symbol),
   762→   762→    );
   763→   763→
   764→   764→    for (const signal of signals) {
   765→   765→      if (!(TREND_TRADE.CONFIRMED_PHASES as readonly string[]).includes(signal.phase)) continue; // 候选趋势不开仓
   766→   766→      if (!signal.dataFresh) {
   767→   767→        this.pushSkip(signal.symbol, '数据不可用/过期（不开新仓）', signal.trendCycleId ?? undefined);
   768→   768→        continue;
   769→   769→      }
   770→   770→      const cycleId = signal.trendCycleId;
   771→   771→      if (!cycleId) continue;
   772→   772→      const side = signal.phase === 'CONFIRMED_UP' ? 'LONG' : 'SHORT';
   773→   773→
   774→   774→      // 幂等守卫 1：同 symbol 已有 OPEN 关联持仓（防御；正常由周期决策键保证）
   775→   775→      if (openSymbols.has(signal.symbol)) continue;
   776→   776→
   777→   777→      // 本周期已存在待入场记录（ACTIVE/CONVERTED/CANCELED 任一）：不重复创建
   778→   778→      const existing = await this.repo.getPendingByCycle(cycleId);
   779→   779→      if (existing) continue;
   780→   780→
   781→   781→      // 容量守卫（跳过不烧周期：仓位/待入场释放后同一确认趋势仍可创建待入场）
   782→   782→      if (eq.openCount + pendingSymbols.size >= cfg.maxPositions) {
   783→   783→        this.recordSkip(signal.symbol, `已达最大持仓数 ${cfg.maxPositions}`, cycleId);
   784→   784→        continue;
   785→   785→      }
   786→   786→
   787→   787→      // 同币种冷却（§5.4）：平仓后 5 分钟内新信号只记录不执行；到期后重走完整门槛
   788→   788→      const lastClose = await this.repo.lastCloseAt(signal.symbol);
   789→   789→      if (lastClose != null && Date.now() - lastClose < TREND_TRADE.COOLDOWN_MS) {
   790→   790→        this.recordSkip(
   791→   791→          signal.symbol,
   792→   792→          `同币种冷却中（距上次平仓 ${Math.ceil((Date.now() - lastClose) / 1000)}s / ${TREND_TRADE.COOLDOWN_MS / 1000}s）`,
   793→   793→          cycleId,
   794→   794→        );
   795→   795→        continue;
   796→   796→      }
   797→   797→
   798→   798→      // 「先平后开」独立性：确认评估包含的已收盘 K 线必须晚于该 symbol 上一笔平仓
   799→   799→      // （同一根 K 线不既触发平仓又触发反向开仓；首笔无平仓记录直接放行）
   800→   800→      if (TREND_TRADE.REVERSE_INDEPENDENT_BARS && lastClose != null) {
   801→   801→        if (signal.lastBarTime != null && signal.lastBarTime + 60_000 <= lastClose) {
   802→   802→          this.recordSkip(signal.symbol, '等待平仓后下一根已收盘 K 线确认（先平后开）', cycleId);
   803→   803→          continue;
   804→   804→        }
   805→   805→      }
   806→   806→
   807→   807→      // 入场质量门槛（评分/ER/高周期过滤）——不满足只记录原因，不创建待入场
   808→   808→      const gateReason = this.entryGateCheck(signal);
   809→   809→      if (gateReason) {
   810→   810→        this.recordSkip(signal.symbol, `入场门槛未过：${gateReason}`, cycleId);
   811→   811→        continue;
   812→   812→      }
   813→   813→
   814→   814→      // 创建待入场记录（每周期一条，数据库主键兜底幂等）；等待下一根已收盘 K 线复核
   815→   815→      const claimed = await this.repo.claimPending({
   816→   816→        cycleId,
   817→   817→        symbol: signal.symbol,
   818→   818→        side,
   819→   819→        firstBarTime: signal.lastBarTime ?? Date.now(),
   820→   820→        firstSeenAt: Date.now(),
   821→   821→        snapshotJson: JSON.stringify(signal),
   822→   822→        status: 'ACTIVE',
   823→   823→        cancelReason: null,
   824→   824→        openPositionId: null,
   825→   825→        createdAt: Date.now(),
   826→   826→        updatedAt: Date.now(),
   827→   827→      });
   828→   828→      if (claimed) {
   829→   829→        pendingSymbols.add(signal.symbol); // 同轮循环内容量即时收紧
   830→   830→        logDecision(
   831→   831→          `待入场创建（首次确认，等待下一根已收盘 K 线复核）symbol=${signal.symbol} side=${side} ` +
   832→   832→            `posCycle=${cycleId} firstBar=${signal.lastBarTime} score=${signal.score.toFixed(0)} ` +
   833→   833→            `er=${signal.features.efficiencyRatio.toFixed(2)} htf5m=${signal.htf5m.dir} htf15m=${signal.htf15m.dir}`,
   834→   834→        );
   835→   835→      }
   836→   836→    }
   837→   837→  }

   992→   992→   * ① 先按标记价正常平掉全部 OPEN 持仓（走引擎平仓：释放保证金、记录流水与 MANUAL 审计）；

   994→   994→   * ③ 清空趋势关联与决策审计（TrendTradeMeta/TrendAutoDecision/TrendPendingEntry——
   995→   995→   *    仅趋势自动账户使用，不带账户维度，全表清空安全）；
   996→   996→   * ④ 余额 = initialCapital（100）、totalDeposited = 0（ROI 本金基准 = initialCapital +
   997→   997→   *    totalDeposited，回到 100）、权益峰值归零；
   998→   998→   * ⑤ 运行模式与持仓数量等配置保持不变。手动盘 default 账户完全不受影响。
   999→   999→   */
  1000→  1000→  async resetAccount(): Promise<{ closed: number }> {
  1001→  1001→    const r = await this.closeAllManual();
  1002→  1002→    const acc = await db.paperAccount.findUnique({ where: { name: TREND_TRADE.ACCOUNT_NAME } });
  1003→  1003→    if (acc) {
  1004→  1004→      await db.paperPosition.deleteMany({ where: { accountId: acc.id } });
  1005→  1005→      await db.paperTrade.deleteMany({ where: { accountId: acc.id } });
  1006→  1006→    }
  1007→  1007→    await db.$executeRawUnsafe(`DELETE FROM TrendTradeMeta`);
  1008→  1008→    await db.$executeRawUnsafe(`DELETE FROM TrendAutoDecision`);
  1009→  1009→    await db.$executeRawUnsafe(`DELETE FROM TrendPendingEntry`);
  1010→  1010→    const cfg = await this.repo.loadConfig(TREND_TRADE.ACCOUNT_NAME);
  1011→  1011→    const capital = cfg?.initialCapital ?? TREND_TRADE.INITIAL_CAPITAL;
  1012→  1012→    if (acc) {
  1013→  1013→      await db.paperAccount.update({
